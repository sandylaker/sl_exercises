\begin{frame}{Exercise 1: L0 Regularization}
	\small
	Consider the regression learning setting, i.e., $\Yspace = \mathbb{R}$, and the feature space $\Xspace = \mathbb{R}^p$. Let the hypothesis space be the linear models: $$\Hspace = \{ \fx = \thetav^T \xv \ | \ \thetav \in \mathbb{R}^p \}.$$
	
	Suppose your loss function of interest is the L2 loss $L(y, \fx) = \frac{1}{2} (y - \fx)^2$. Consider the $L_0$-regularized empirical risk of a model $\fxt$: 
	
	$$
		\riskrt = \risket + \lambda ||\thetav ||_0 = \frac{1}{2} \sum_{i=1}^n (\yi - \thetav^T \xi )^2 + \lambda \sum_{i=1}^p \thetainnz.
	$$
	
	Assume that $\Xmat^T \Xmat = \Imat$, which holds if $\Xmat$ has orthonormal columns. Show that the minimizer $\thetah_{L0} = (\thetah_{L0, 1}, \ldots, \thetah_{L0, p})^T$ is given by 
	$$
		\thetah_{L0, i} = \thetahi \id_{|\thetahi| > \sqrt{2 \lambda}}, \qquad i = 1, \ldots, p,
	$$
	where $\thetavh = (\thetah_1, \ldots, \thetah_p)^T = (\Xmat^T \Xmat)^{-1} \Xmat^T \yv$ is the minimizer of the unregularized empirical risk. For this purpose, using the following steps:

\end{frame}

\begin{frame}{Exercise 1 (i)}
	(i) Derive that 
	\begin{align*}
		\argmin_{\thetav} \risket = \argmin_{\thetav} \sumip - \thetahi \thetai + \frac{\thetai^2}{2} + \lambda \thetainnz.
	\end{align*}
	
	Note that $\thetai$ is from the minimizer $\thetavh$ of the unregularized empirical risk :
	\begin{align*}
		\thetavh = (\Xmat^T \Xmat)^{-1} \Xmat^T \yv = \Xmat^T \yv
	\end{align*}
	because we assume that $\Xmat^T \Xmat = \Imat$.
\end{frame}

\begin{frame}{Solution to Exercise 1 (i)}
	\begin{align*}
		\argmin_{\thetav} \riskr(\thetav) &= \argmin_{\thetav} \frac{1}{2} ||\Xmat\thetav - \yv||_2^2  + \lambda \sumip \thetainnz \\
		\only<2->{
		&= \argmin_{\thetav} \frac{1}{2} \yv^T \yv - \yv^T \Xmat \thetav + \frac{1}{2} \thetav^T \Xmat^T \Xmat \thetav + \lambda \sumip \thetainnz \\}
		\only<3->{&= \argmin_{\thetav} - \underbrace{\yv^T \Xmat}_{\thetavh^T} \thetav + \frac{1}{2} \thetav^T \underbrace{\Xmat^T \Xmat}_{= \Imat} \thetav + \lambda \sumip \thetainnz \\}
		\only<4->{&= \argmin_{\thetav} - \thetavh^T \thetav + \frac{1}{2} \thetav^T \thetav + \lambda \sumip \thetainnz \\}
		\only<5->{&= \argmin_{\thetav} \sumip - \thetahi \thetai + \frac{\thetai^2}{2} + \lambda \sumip \thetainnz \\}
	\end{align*}
\end{frame}

\begin{frame}{Exercise 1 (ii)}
	(ii) Note that the minimization problem on the right-hand side of (i) can be written as $\sumip g_i(\theta)$, where
		$$g_i(\theta) = -\thetahi \theta + \frac{\theta^2}{2} + \lambda \id_{|\theta| \neq 0}.$$
		What is the advantage of this representation if we seek to find the $\thetav$ with entries $\theta_1, \ldots, \theta_p$ minimizing $\riskr(\thetav)$?

	\bigskip
	\begin{center}
	{\setlength{\fboxsep}{7pt}\colorbox{black!8}{\parbox{0.92\textwidth}{
		\footnotesize\textbf{Recall from (i):}
		\[
			\argmin_{\thetav} \risket \;=\; \argmin_{\thetav} \sum_{i=1}^p \left( -\thetahi \theta_i + \frac{\theta_i^2}{2} + \lambda \id_{|\theta_i| \neq 0} \right).
		\]
	}}}
	\end{center}
\end{frame}


\begin{frame}{Solution to Exercise 1 (ii)}
	\small
	\textbf{Advantage:} minimize each $g_i$ \textbf{separately}.

	\smallskip
	\begin{columns}[T]
		\column{0.55\textwidth}
		\textbf{Why?}
		\begin{itemize}\setlength{\itemsep}{2pt}
			\item Each $g_i$ involves \emph{only} $\theta_i$ (not $\theta_j$, $j \neq i$).
			\item Coordinates \emph{decoupled}: moving $\theta_i$ doesn't touch any $g_j$.
		\end{itemize}

		\smallskip
		\textbf{Key identity}\,---\,min of sum $=$ sum of mins:
		\[
			\min_{\theta_1, \ldots, \theta_p} \sum_{i=1}^p g_i(\theta_i) \;=\; \sum_{i=1}^p \min_{\theta_i} g_i(\theta_i).
		\]

		\smallskip
		\textbf{Payoff:} $p$-D problem $\rightarrow$ $p$ 1-D problems; \\ solve $\min_\theta g_i(\theta)$ once, parametric in $\thetahi$.

		\column{0.42\textwidth}
		\centering
		{\scriptsize \textbf{Decoupled subproblems}}\\[0.5em]
		\begin{tikzpicture}[scale=0.7, font=\footnotesize]
			\node[draw, rounded corners, fill=blue!8, minimum width=1.05cm, minimum height=0.7cm] (g1) at (0,0) {$g_1(\theta_1)$};
			\node[draw, rounded corners, fill=blue!8, minimum width=1.05cm, minimum height=0.7cm] (g2) at (1.7,0) {$g_2(\theta_2)$};
			\node at (3.0,0) {$\cdots$};
			\node[draw, rounded corners, fill=blue!8, minimum width=1.05cm, minimum height=0.7cm] (gp) at (4.3,0) {$g_p(\theta_p)$};
			\draw[->,thick,gray!70] (g1.south) -- node[right=1pt,gray,font=\scriptsize] {$\min_\theta$} ++(0,-0.7) node[below=2pt,black] {$\theta_1^*$};
			\draw[->,thick,gray!70] (g2.south) -- ++(0,-0.7) node[below=2pt,black] {$\theta_2^*$};
			\draw[->,thick,gray!70] (gp.south) -- ++(0,-0.7) node[below=2pt,black] {$\theta_p^*$};
		\end{tikzpicture}\\[0.5em]
		{\scriptsize \textcolor{black!60}{Each $g_i$ optimised in parallel,\\independently of the others.}}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (iii)}

	\bigskip
	\emph{Hint:} Show that $g_i(\thetai) < 0 = g_i(0)$ and argue that the minimizer must have the same sign as $\thetahi$.\\

\begin{frame}{Solution to Exercise 1 (iii)}
	\small
	\begin{center}
	{\setlength{\fboxsep}{6pt}\colorbox{black!8}{\parbox{0.92\textwidth}{
		\centering\footnotesize
		\textbf{Objective:}\ \ $g_i(\thetai) = -\thetahi\,\thetai + \dfrac{\thetai^2}{2} + \lambda\,\id_{|\thetai|\neq 0}.$
	}}}
	\end{center}

	\smallskip
	\begin{columns}[T]
		\column{0.50\textwidth}
		\textbf{Split $g_i(\thetai)$ at the jump $\thetai = 0$.}
		\begin{itemize}\setlength{\itemsep}{6pt}
			\item<2-> \textbf{Case 1:}\ $\thetai = 0$ \\
				\(\displaystyle g_i(0) = 0.\)
			\item<3-> \textbf{Case 2:}\ $\thetai \neq 0$, so $g_i$ is a smooth quadratic
				\[ g_i(\thetai) = \frac{\thetai^2}{2} - \thetahi \thetai + \lambda. \]
				Minimizer $\thetai^* = \thetahi$, with min value
				\[ g_i(\thetahi) = -\frac{\thetahi^2}{2} + \lambda. \]
		\end{itemize}

		\column{0.46\textwidth}
		\uncover<4->{
			\textbf{Why compare?} The penalty $\lambda\,\id_{|\thetai|\neq 0}$ makes $g_i$ \emph{jump} at $0$\,---\,each case yields its own candidate minimum, so the global min is whichever is smaller.

			\medskip
			\textbf{Compare} using $|\thetahi|^2 > 2\lambda$:
			\[
				\underbrace{-\frac{\thetahi^2}{2} + \lambda}_{\text{Case 2 min}} \;<\; \underbrace{0\strut}_{=\,g_i(0)}
			\]
		}

		\uncover<5->{
			\medskip
			\textbf{Conclusion:}\quad \fbox{$\thetai^* = \thetahi$\ \ when $|\thetahi| > \sqrt{2\lambda}$}.
		}
	\end{columns}
\end{frame}

\begin{frame}{Putting it all together}
	\small
	From (iii) and (v), per coordinate $i$:

	\smallskip
	\textit{Strong:}\quad $|\thetahi| > \sqrt{2\lambda} \;\Longrightarrow\; \thetai^* = \thetahi$ \quad (here $\id_{|\thetahi|>\sqrt{2\lambda}} = 1$).

	\smallskip
	\textit{Weak:}\quad $|\thetahi| \leq \sqrt{2\lambda} \;\Longrightarrow\; \thetai^* = 0$ \quad (here $\id_{|\thetahi|>\sqrt{2\lambda}} = 0$).

	\uncover<2->{
		\bigskip
		Glue via the indicator (step (iv)):
		\[
			\thetai^* = \thetahi \cdot \id_{|\thetahi|>\sqrt{2\lambda}}, \quad i = 1, \ldots, p.
		\]
	}

	\uncover<3->{
		\medskip
		By separability (step (ii)), the joint minimizer assembles coordinate-wise:
		\[
			\boxed{\;\thetah_{L0,i} \;=\; \thetah_i \cdot \id_{|\thetah_i|>\sqrt{2\lambda}}, \quad i = 1, \ldots, p\;} \qquad \blacksquare
		\]
	}
\end{frame}

\begin{frame}{Exercise 2 (a): simulate the data}
	\structure{\textit{Task.}}\ \ Build a data set with one informative feature buried in noise.

	\bigskip
	Draw $n = 100$ observations from
	\[
		Y = \sin(x_1) + \varepsilon, \qquad \varepsilon \sim \mathcal{N}(0, \sigma^2),
	\]
	so the response depends on $x_1$ \emph{only}.

	\bigskip
	Add $p = 100$ further covariates $x_2, \ldots, x_{101}$ that are unrelated to $Y$.

	\bigskip
	{\setlength{\fboxsep}{6pt}\colorbox{black!8}{\parbox{0.92\textwidth}{\centering\footnotesize
	$101$ candidate features, but $100$ of them are pure noise $\;\Longrightarrow\;$ the regime where regularization earns its keep.}}}
\end{frame}

\begin{frame}[fragile]{Exercise 2 (a): implementation (Python)}
	\begin{lstlisting}[style=pystyle]
import numpy as np

rng = np.random.default_rng(42)
n, p_add = 100, 100

# feature matrix: n rows, (p_add + 1) columns
X = rng.standard_normal((n, p_add + 1))

# only the first column drives Y; the rest are pure noise
Y = np.sin(X[:, 0]) + rng.normal(scale=0.5, size=n)
\end{lstlisting}

	\smallskip
	\footnotesize
	\structure{$\blacktriangleright$}\ \ \texttt{X} has shape $100 \times 101$; only column \texttt{0} is signal, columns \texttt{1--100} are noise. The Gaussian draw fixes the unspecified $\varepsilon$ to $\mathcal{N}(0, 0.5^2)$.
\end{frame}

from sklearn.linear_model import LinearRegression

def poly_fit(x, y, degree, x_eval):

\begin{frame}[fragile]{Exercise 2 (b): the underdetermined problem}
	With $n = 100 < p + 1 = 101$, the Gram matrix $\Xmat^\top\Xmat$ is singular, so the normal equations have no unique solution.
	\bigskip
\begin{lstlisting}[style=pystyle]
XtX = X.T @ X                          # shape (101, 101)
np.linalg.matrix_rank(XtX)             # -> 100  (< 101)
np.linalg.cond(XtX)                    # -> 2.0e+17
beta = np.linalg.solve(XtX, X.T @ Y)   # numerically meaningless
\end{lstlisting}
	\smallskip
	\footnotesize
	R's \texttt{solve} aborts with ``system is computationally singular''; NumPy returns a vector, but the condition number $\sim 10^{17}$ flags it as garbage.
\end{frame}

\begin{lstlisting}[style=pystyle, basicstyle=\ttfamily\tiny]
from sklearn.neural_network import MLPRegressor
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error

Xtr, Xv, ytr, yv = train_test_split(
    X, Y, test_size=0.2, random_state=42)
net = MLPRegressor((50, 50), activation="relu",
        solver="adam", warm_start=True, max_iter=1,
        random_state=42)

best, wait, patience = np.inf, 0, 50
for epoch in range(300):
    net.fit(Xtr, ytr)             # one epoch
    v = mean_squared_error(yv, net.predict(Xv))
    if v < best - 1e-4:
        best, wait = v, 0
    else:
        wait += 1
        if wait >= patience:
            break                 # stop early
\end{lstlisting}
