\begin{frame}{Exercise 1: Soft Margin Classifier}

The primal optimization problem for the two-class soft margin SVM classification is given by 
\begin{align*}
	\min_{\thetav, \theta_0, \zetai} & \frac{1}{2} ||\thetav||^2 + \sumin \zetai \\	
	\text{s.t.: }\quad & \yi (\thetav^T \xi + \theta_0) \geq 1 - \zetai, \\
	& \zetai \geq 0, \qquad \forall i= 1, \ldots, n.
\end{align*}

\begin{frame}{Solution to Exercise 1 (b): point $(0,0)$}
	At the optimum $\thetavh = (1,1)^T,\ \thetah_0 = -2 \;\Rightarrow\; f(\xv) = x_1 + x_2 - 2$, \; so $m^{(i)} = \yi\,(x_1^{(i)} + x_2^{(i)} - 2)$.
	\bigskip
	\begin{itemize}
		\setlength{\itemsep}{12pt}
		\item<2-> $(0,0),\ \yi[1] = +1$: \quad $m^{(1)} = 1\,(0 + 0 - 2) = -2 \;\;(< 1$, violates margin$)$
		\item<3-> apply the recipe: \; $\sli[1] = \max\!\big(0,\, 1 - m^{(1)}\big) = \max(0,\, 3) = $ \colorbox{blue!10}{$3$}
	\end{itemize}
\end{frame}

\begin{frame}{Solution to Exercise 1 (b): remaining points}
	Same recipe: $\;m^{(i)} = \yi\,(x_1^{(i)} + x_2^{(i)} - 2)$, \quad $\sli = \max(0,\, 1 - m^{(i)})$.
	\smallskip
	\begin{itemize}
		\setlength{\itemsep}{5pt}
		\item<2-> $(0.5, 0.5),\; y = -1$: \; $m^{(2)} = -1\,(0.5 + 0.5 - 2) = 1 \;\Rightarrow$ on margin, $\sli[2] = 0$
		\item<3-> $(0, 1),\; y = -1$: \; $m^{(3)} = -1\,(0 + 1 - 2) = 1 \;\Rightarrow$ on margin, $\sli[3] = 0$
		\item<4-> $(0, 3),\; y = +1$: \; $m^{(4)} = 1\,(0 + 3 - 2) = 1 \;\Rightarrow$ on margin, $\sli[4] = 0$
		\item<5-> $(3, 0),\; y = +1$: \; $m^{(5)} = 1\,(3 + 0 - 2) = 1 \;\Rightarrow$ on margin, $\sli[5] = 0$
		\item<6-> $(3, 3),\; y = +1$: \; $m^{(6)} = 1\,(3 + 3 - 2) = 4 > 1 \;\Rightarrow$ beyond margin, $\sli[6] = 0$
	\end{itemize}
	\medskip
	\uncover<7->{\colorbox{blue!10}{\begin{minipage}{0.95\textwidth}\textbf{On margin:} $(0.5,0.5),\,(0,1),\,(0,3),\,(3,0)$. \quad \textbf{Violator:} $(0,0)$. \quad \textbf{Off margin:} $(3,3)$.\end{minipage}}}
\end{frame}

\begin{frame}[fragile]{Solution to Exercise 2: PEGASOS}
	Unconstrained regularized ERM with the hinge loss $\max(0,\,1-\yi\fxi)$ from (b). \\
	SGD step: shrink $\thetav$ (the penalty); if a point is inside the margin, add the hinge subgradient.
	\begin{lstlisting}
def pegasos_linear(y, X, nr_iter=50000, lam=1.0, alpha=0.01, seed=0):
    # soft-margin SVM in the primal via stochastic subgradient descent
    # X: design matrix with leading 1s column (intercept); y in {-1,+1}
    rng = np.random.default_rng(seed)
    n, p = X.shape
    theta = rng.standard_normal(p)
    for _ in range(nr_iter):
        i = rng.integers(n)
        f_i = X[i] @ theta                 # margin score
        theta = (1 - lam * alpha) * theta  # weight decay (L2 penalty)
        if y[i] * f_i < 1:                 # inside margin: subgradient step
            theta = theta + alpha * y[i] * X[i]
    return theta
	\end{lstlisting}
\end{frame}
