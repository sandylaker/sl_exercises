\begin{frame}{Exercise 1: Gradient Boosting}
	In the following, you assume that your outcome follows a $\log_2$-normal distribution with density function
	$$p(y|f) = \frac{1}{y \sigma \sqrt{2\pi}} \exp\left( - \frac{(\log_2(y) - f)^2}{2\sigma^2} \right)$$
	where $\sigma = 1$. In other words, $\log_2(Y)$ follows a normal distribution. You observe $n = 3$ data points $\yv$ and want to model $f$ using features $\Xmat \in \mathbb{R}^{n \times p}$. You choose to use gradient boosting with tree base learners. (The concrete data and the $p = 2$ features are specified in part (b).)

	(a) Derive the pseudo residuals based on the negative log-likelihood for the given distribution assumption.
\end{frame}

\begin{frame}{Gradient Boosting: One Iteration at a Glance}
	\small
	Parts (a)--(c) walk through the loop below, repeated for $m = 1, \dots, M$ rounds. At round $m$ we fit a base learner to the \emph{current} pseudo-residuals, then take an additive step.
	\medskip
	\begin{center}
	\begin{tikzpicture}[font=\scriptsize, >=Latex,

\begin{frame}{1 (a): Think Before Solving}
	\small
	\textbf{Target:}\quad turn the distribution $p(y|f)$ into a pseudo-residual $\tilde r$.
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{Likelihood.} The loss in gradient boosting is the NLL $L = -\ell$ with $\ell = \log p(y|f)$ --- what does $\ell$ look like once you take the log?
		\item<2-> \textbf{Constants.} Which terms do not depend on $f$? Why may they be dropped for the gradient?
		\item<3-> \textbf{Negative gradient.} Differentiate the squared term $\tfrac12(\log_2(y)-f)^2$ --- and recall the pseudo-residual is the \emph{negative} gradient $\tilde r = -\partial L / \partial f$.
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{$p(y|f) \to \ell(f) \to L = -\ell \to \tilde r = -\partial L/\partial f$}}
\end{frame}

\begin{frame}{Solution to Exercise 1 (a)}
	\small
	(a) Derive the pseudo residuals based on the negative log-likelihood.
	\setlength{\jot}{3pt}
	\begin{itemize}
		\setlength{\itemsep}{8pt}
		\item<1-> Log-likelihood (drop $f$-independent constants):
			$$\ell(f) = \log p(y|f) = \text{const} - \tfrac12 (\log_2(y) - f)^2.$$
		\item<2-> The loss is the negative log-likelihood:
			$$L(y, f) = - \ell(f) = \text{const} + \tfrac12 (\log_2(y) - f)^2.$$
		\item<3-> The pseudo-residual is the \emph{negative} gradient of the loss:
			$$\tilde{r} = - \frac{\partial L(y,f)}{\partial f} = \frac{\partial \ell(f)}{\partial f} = \log_2(y) - f. \hfill\qed$$
	\end{itemize}
\end{frame}

\begin{frame}{Exercise 1 (b)}
	\small
	(b) Given only the 3 samples $\yv = (1, 2, 4)^\top$

	and two features
	$$\Xmat = (\xv_1, \xv_2) =
		\begin{pmatrix}
			1 & 0 \\
			1 & 0 \\
			0 & 0 \\
		\end{pmatrix}
	$$
	explicitly derive or state with explanation
	\begin{enumerate}[(i)]
		\item the loss-optimal initial boosting model $\fhx{0}$,
		\item the pseudo residual $\rtilde{1}$,

\end{frame}

		\item<3-> Given the current model, how are the residuals $\tilde r$ formed (cf.\ (a))?

\begin{frame}{1 (b)(i): Loss-optimal Initial Model $\fhx{0}$}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	\begin{itemize}
		\setlength{\itemsep}{10pt}
		\item<1-> Initialize with the loss-minimizing constant:
			$$\fhx{0} = \argmin_{f^{[0]}} \sumin L(\yi, f^{[0]}(\xi)).$$
		\item<2-> Since $L$ is the squared error on $\log_2 y$, the minimizer is the mean:
			$$\fhx{0} = \frac{1}{n} \sumin \log_2(\yi) = \frac{0+1+2}{3} = 1.$$
		\item<3-> Hence the initial predictions on all 3 samples are
			$$\fhv{0} = (1, 1, 1)^\top. \hfill\qed$$
	\end{itemize}
	\end{column}
	\begin{column}{0.25\textwidth}
	\rmgb{1}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (b)(ii): Pseudo Residual $\rtilde{1}$}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	\begin{itemize}
		\setlength{\itemsep}{8pt}
		\item<1-> From (a): $\tilde{r}(f) = -\partial L / \partial f = \log_2(y) - f$.
		\item<2-> Current predictions: $\fhv{0} = \left(\fh^{[0]}(\xv^{(1)}), \fh^{[0]}(\xv^{(2)}), \fh^{[0]}(\xv^{(3)})\right)^\top = (1, 1, 1)^\top$.
		\item<3-> So the first-round residuals are
			$$\rtilde{1} = \log_2\yv - \fhv{0}
			= \begin{pmatrix} 0 \\ 1 \\ 2 \end{pmatrix} - \begin{pmatrix} 1 \\ 1 \\ 1 \end{pmatrix}
			= \begin{pmatrix} -1 \\ 0 \\ 1 \end{pmatrix}. \hfill\qed$$
	\end{itemize}
	\end{column}
	\begin{column}{0.25\textwidth}
	\rmgb{2}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (b)(iii): Regression Stump $\Rt{1}$, $t = 1, 2$}
	\small
	\begin{columns}[T]
	\begin{column}{0.50\textwidth}
	Fit the stump to $\rtilde{1} = (-1, 0, 1)^\top$:
	\begin{itemize}
		\setlength{\itemsep}{5pt}
		\item<1-> Split on $x_1$ ($x_2$ carries no info); note $x_1^{(1)} = x_1^{(2)}$.
		\item<2-> Each leaf value $=$ mean residual of its samples.

	\end{itemize}
	\uncover<5->{
	\medskip

	\node[leaf, fill=orange!12, draw=orange!60!black] (rr) at (1.05,-1.8) {$x_1 < 0.5$\\ \{3\}\\ $+1$};

\begin{frame}{1 (b)(iv): Boosting Model $\fhx{1}$ (predictions $\fhv{1}$)}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	\begin{itemize}
		\setlength{\itemsep}{7pt}
		\item<1-> Recall $R_1^{[1]} = -0.5\,\id_{x_1 \ge 0.5}$, $R_2^{[1]} = 1\cdot\id_{x_1 < 0.5}$, and $\alpha = 1$.
		\item<2-> Update direction (per sample) from the stump: $(-0.5, -0.5, 1)^\top$.
		\item<3-> Additive update:
			\begin{align*}
				\fhv{1} &= \fhv{0} + \alpha\cdot(-0.5, -0.5, 1)^\top \\
				&= \begin{pmatrix} 1 \\ 1 \\ 1 \end{pmatrix} + \begin{pmatrix} -0.5 \\ -0.5 \\ 1 \end{pmatrix}
				= \begin{pmatrix} 0.5 \\ 0.5 \\ 2 \end{pmatrix}. \hfill\qed
			\end{align*}
	\end{itemize}
	\end{column}
	\begin{column}{0.25\textwidth}
	\rmgb{4}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (b)(v): Pseudo Residual $\rtilde{2}$}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	\begin{itemize}
		\setlength{\itemsep}{8pt}
		\item<1-> Same computation as (ii), now against the updated model $\fhv{1}$.
		\item<2-> So
			\begin{align*}
				\rtilde{2} &= \log_2\yv - \fhv{1} \\
				&= \begin{pmatrix} 0 \\ 1 \\ 2 \end{pmatrix} - \begin{pmatrix} 0.5 \\ 0.5 \\ 2 \end{pmatrix}

\begin{frame}{Solution to Exercise 1 (c)}
	\small
	(c) What would happen in the second iteration?
	\begin{columns}[T]
	\begin{column}{0.58\textwidth}
	\begin{itemize}
		\setlength{\itemsep}{6pt}
		\item<2-> Fit $\rtilde{2} = (-0.5, 0.5, 0)^\top$ with $R_1^{[2]}, R_2^{[2]}$.
		\item<3-> Split on $x_1$: samples 1,2 $\to R_1^{[2]}$, sample 3 $\to R_2^{[2]}$.
		\item<4-> Leaf values:
			$$R_1^{[2]} = \tfrac{-0.5 + 0.5}{2}\,\id_{x_1 \ge 0.5} = 0, \quad R_2^{[2]} = 0\cdot\id_{x_1 < 0.5} = 0.$$
		\item<5-> Update direction $(0,0,0)^\top$ $\Rightarrow$ \textbf{boosting stalls}: these features cannot improve the model further.
	\end{itemize}
	\end{column}
	\begin{column}{0.40\textwidth}
	\uncover<4->{
	\vtop{\centering
	\begin{tikzpicture}[font=\scriptsize]
	\node[draw, rounded corners=2pt, fill=green!10, draw=green!45!black, align=center, minimum width=3.5cm, inner sep=4pt] (box) at (0,0)
		{leaf $x_1 \ge 0.5$: samples 1,2\\[3pt]
		identical features $x_1 = 1$\\[3pt]
		residuals $-0.5$ and $+0.5$\\[3pt]
		$\Rightarrow$ average $= 0$ (cancel)};
	\end{tikzpicture}
	\smallskip\\
	\parbox{3.5cm}{\centering\scriptsize Opposite-sign residuals on indistinguishable samples cancel.}}
	}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (d)}
	(d) If you are given more data points, but still the two binary feature vectors $\xv_1$ and $\xv_2$, what will happen as
	\begin{enumerate}[(i)]
		\item the number of boosting rounds $M$ grows
		\item the sample size $n$ grows
	\end{enumerate}
	in terms of model capacity (if $d$ is kept fixed)?

	\vspace{8pt}
	\uncover<2->{\begin{center}\setlength{\fboxsep}{6pt}\colorbox{blue!10}{\begin{minipage}{0.85\linewidth}\footnotesize\textbf{Leading question:} which knob adds new base learners to the additive model --- the number of rounds $M$, or the sample size $n$?\end{minipage}}\end{center}}
\end{frame}

\begin{frame}{Solution to Exercise 1 (d)}
	\small
	\begin{columns}[T]
	\begin{column}{0.60\textwidth}
	\begin{enumerate}[(i)]
		\setlength{\itemsep}{10pt}
		\item<1-> \textbf{$M$ grows:} each round adds another stump (write $b^{[m]}$ for the round-$m$ stump) to $\fh = \fh^{[0]} + \sum_{m=1}^{M} \alpha\, b^{[m]}$ $\Rightarrow$ capacity \emph{increases}, the algorithm may \textbf{overfit}.
		\item<2-> \textbf{$n$ grows:} the number of base learners (depth $d$ fixed, $M$ fixed) is unchanged $\Rightarrow$ capacity \emph{stays the same}, the algorithm may \textbf{underfit}.
	\end{enumerate}
	\end{column}
	\begin{column}{0.38\textwidth}
	\uncover<2->{
	\centering
	\begin{tikzpicture}[font=\scriptsize, >=Latex,
		k/.style={draw, rounded corners=2pt, align=center, inner sep=4pt, minimum width=2.4cm}]
	\node[k, fill=blue!10, draw=blue!55!black] (M) at (0,0) {$M \uparrow$\\ more stumps\\ capacity $\uparrow$};
	\node[k, fill=black!4, draw=black!35] (n) at (0,-1.8) {$n \uparrow$\\ same \# stumps\\ capacity fixed};
	\end{tikzpicture}
	}
	\end{column}
	\end{columns}
\end{frame}
