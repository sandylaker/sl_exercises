\begin{frame}{Exercise 1}
	For a design matrix $\Xmat \in \mathbb{R}^{n \times p}$ and the vector of targets $\yv \in \mathbb{R}^n$, consider Lasso regression, i.e., 
	$$\argmin_{\thetav \in \mathbb{R}^p} 0.5 || \Xmat \thetav - \yv ||_2^2 + \lambda ||\thetav||_1.$$
	where $\lambda > 0$ is the regularization parameter.
	
	(a) Since there exists no analytical solution to Lasso regression in general, we want to find a procedure similar to gradient descent that should converge to the true solution.
	
	\vspace{10pt}
\end{frame}

\begin{frame}{Exercise 1 (a) (i)}
	\textbf{Question (a) (i)}: Explain why $\riskr$ is not differentiable.
	
	\vspace{10pt}
	\uncover<2->{
		\textbf{Solution}: Because $\lambda ||\thetav||_1$ is not differentiable at $\thetav = \zero$.
	}
\end{frame}

\begin{frame}{Exercise 1 (a) (ii)}
	Recall the regularized risk of the Lasso problem $$\riskr = 0.5 ||\Xmat \thetav - \yv ||_2^2 + \lambda ||\thetav||_1.$$

	\textbf{Question (a) (ii)}: Show that $\riskr$ is convex. \emph{Hint: The sum of convex function is convex.}
	
	\vspace{10pt} 
	
	\textbf{Solution}: 
		\begin{enumerate}
			\item<2-> $0.5 ||\Xmat \thetav - \yv ||_2^2$ is convex, since it is quadratic.
			\item<3-> $\lambda || \thetav ||_1$ is also convex (since it is a norm).
			\item<4-> Sum of convex functions is convex. So $\riskr$ is convex.
		\end{enumerate}
\end{frame}

\begin{frame}{Exercise 1 (a) (iii)}
	\small
	\textbf{Question (a) (iii)}: Find $\rho_j, z_j \in \mathbb{R}$ for which $$0.5 \frac{\partial}{\partial \theta_j} ||\Xmat \thetav - \yv ||_2^2 = -\rho_j + \theta_j z_j.$$
	\textbf{Solution}: 
	\begin{align*}
		\only<2->{
			0.5 \frac{\partial}{\partial \theta_j} ||\Xmat \thetav - \yv ||_2^2
		} 
		\only<3->{
			&= \frac{\partial}{\partial \theta_j} 0.5 \sumin \left(\yi - \sum_{k=1}^p \xik \theta_k \right)^2 \\
		}
		\only<4->{
			&= - \sumin \xij \left( \yi - \sum_{k=1}^p \xik \theta_k \right) \\
		}
		\only<5->{
			&= - \underbrace{\sumin \xij \left( \yi - \sum_{k\neq j}^p \xik \theta_k \right)}_{= \rho_j} + \theta_j \underbrace{\sumin \left( \xij \right)^2}_{ = z_j}
		}
	\end{align*}
\end{frame}

\begin{frame}[t]{Exercise 1 (a) (iii): matrix form, step by step}
	\footnotesize
	\textbf{Question (a)(iii):} find $\rho_j, z_j$ with $0.5\,\tfrac{\partial}{\partial\theta_j}\|\Xmat\thetav-\yv\|_2^2 = -\rho_j + \theta_j z_j$.
	\smallskip
	\begin{columns}[T]
	\begin{column}{0.55\textwidth}
		\only<2->{\textbf{Step 1.} residual $\mathbf{r} := \Xmat\thetav - \yv$, gradient $\Xmat^\top\mathbf{r}$; component $j =$ column $\Xmat_{\cdot j}$ dotted with $\mathbf{r}$:}
		\only<3->{\[ 0.5\,\tfrac{\partial}{\partial\theta_j}\|\Xmat\thetav-\yv\|_2^2 = \Xmat_{\cdot j}^\top\mathbf{r} \]}

		\smallskip
		\only<4->{\textbf{Step 2.} isolate $\theta_j$ via $\Xmat\thetav = \Xmat_{\cdot j}\theta_j + \Xmat_{-j}\thetav_{-j}$, group:}
		\only<5->{\[ \begin{aligned}
			\Xmat_{\cdot j}^\top\mathbf{r} &= \underbrace{\Xmat_{\cdot j}^\top\Xmat_{\cdot j}}_{z_j}\theta_j \\
			&\quad - \underbrace{\Xmat_{\cdot j}^\top(\yv - \Xmat_{-j}\thetav_{-j})}_{\rho_j} \\
			\onslide<6->{&= -\rho_j + \theta_j z_j}
		\end{aligned} \]}
	\end{column}
	\begin{column}{0.45\textwidth}
		\only<7->{\textbf{Step 3.} The gradient is a line in $\theta_j$, $\,-\rho_j + \theta_j z_j$:}
		\begin{itemize}
			\item<8-> \emph{slope} $z_j = \|\Xmat_{\cdot j}\|_2^2 \ge 0$: size (energy) of feature $j$.
			\item<9-> \emph{intercept} $-\rho_j$: here $\rho_j = \Xmat_{\cdot j}^\top\mathbf{r}_{-j}$ correlates feature $j$ with the partial residual $\mathbf{r}_{-j} = \yv - \Xmat_{-j}\thetav_{-j}$ (target minus the other features' fit).
			\item<10-> \emph{root} $\theta_j = \rho_j/z_j$: best coordinate-$j$ fit (OLS); (a)(v) soft-thresholds it.
		\end{itemize}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}[t]{Exercise 1 (a): what we have so far}
	\small
	Goal: minimize $\riskr(\thetav) = 0.5\|\Xmat\thetav-\yv\|_2^2 + \lambda\|\thetav\|_1$ one coordinate $\theta_j$ at a time.
	\medskip
	\begin{itemize}
		\item (i) $\riskr$ \emph{not} differentiable --- the kink of $\lambda\|\thetav\|_1$ at $\thetav = \zero$.
		\item (ii) but $\riskr$ \emph{convex} --- a per-coordinate minimum is global.
		\item (iii) the smooth part is linear in $\theta_j$: $0.5\,\tfrac{\partial}{\partial\theta_j}\|\Xmat\thetav-\yv\|_2^2 = -\rho_j + \theta_j z_j$, \\ with slope $z_j = \|\Xmat_{\cdot j}\|_2^2$ and $\rho_j = \Xmat_{\cdot j}^\top\mathbf{r}_{-j}$ (partial residual).
	\end{itemize}
	\medskip
	\textbf{Next (a)(iv):} only $\lambda|\theta_j|$ remains, non-smooth at $0$. Use its \emph{subderivative} and add to (iii) to assemble $\partial_{\theta_j}\riskr$.
\end{frame}

\begin{frame}{Exercise 1 (a) (iv)}
	\small
	\textbf{Question (a) (iv)}: In this situation, we can use the so-called sub-derivative which we denote with $\partial f$ for a real-valued convex continuous function $f$. The subderivative maps a point $\theta \in \mathbb{R}$ to and interval
	\begin{itemize}
		\item and if $f$ is differentiable at $\tilde{\theta} \in \mathbb{R}$, then $\partial f(\tilde{\theta}) = \{ \frac{\mathrm{d}}{\mathrm{d}\theta} f(\tilde{\theta}) \}$,
		\item and for $f(\theta) = \lambda |\theta|$ and $\lambda > 0$, it holds that $\partial f(\theta) = \begin{cases}
			\{ -\lambda \} \quad & \text{for } \theta < 0, \\
			[ -\lambda, \lambda] \quad & \text{for } \theta = 0, \\
			\{ \lambda \} \quad & \text{for } \theta > 0
		\end{cases}
		$
		\item and for $f, g$ real-valued convex functions with $\partial f(\tilde{\theta}) = [a, b], \partial g(\tilde{\theta}) = [c, d]$, $$\partial(f+g)(\tilde{\theta}) = [a + c, b + d]$$ where $b \geq a$ and $d \geq c$.
	\end{itemize}
	With this compute the sub-derivative of $\mathcal{R}_{\text{reg}, \thetav_{\neq j}}$ w.r.t. $\theta_j$, i.e., $\partial_{\theta_j} \mathcal{R}_{\text{reg}, \thetav_{\neq j}} $ where $\mathcal{R}_{\text{reg}, \thetav_{\neq j}}: \mathbb{R} \to \mathbb{R}_{\geq 0}, \theta_j \longmapsto \riskr(\theta_1, \ldots, \theta_j, \ldots, \theta_p)$ for constants $\thetav_{\neq j} = (\theta_1, \ldots, \theta_{j-1}, \theta_{j+1}, \ldots, \theta_p)^T$. \emph{Hint: Use (a) (iii).}
\end{frame}

\begin{frame}[t]{Exercise 1 (a) (iv): case by case}
	\setbeamercovered{invisible}
	\textbf{Solution}: by the sum rule $\partial_{\theta_j}\mathcal{R}_{\text{reg},\thetav_{\neq j}} = \underbrace{\{-\rho_j+\theta_j z_j\}}_{\text{from (iii)}} + \partial(\lambda|\theta_j|)$, and $\partial(\lambda|\theta_j|)$ splits on the sign of $\theta_j$:
	\medskip

	\renewcommand{\arraystretch}{1.5}
	\begin{center}
	\begin{tabular}{ccc|c}
		\hline
		$\theta_j$ & smooth part & $\partial(\lambda|\theta_j|)$ & $\partial_{\theta_j}\mathcal{R}_{\text{reg}}$ \\
		\hline
		$< 0$ & $\{-\rho_j+\theta_j z_j\}$ & $\{-\lambda\}$ & \uncover<2->{$\{-\rho_j+\theta_j z_j-\lambda\}$} \\
		$= 0$ & $\{-\rho_j\}$ & $[-\lambda,\lambda]$ & \uncover<3->{$[-\rho_j-\lambda,\ -\rho_j+\lambda]$} \\
		$> 0$ & $\{-\rho_j+\theta_j z_j\}$ & $\{\lambda\}$ & \uncover<4->{$\{-\rho_j+\theta_j z_j+\lambda\}$} \\
		\hline
	\end{tabular}
	\end{center}
	\medskip

	\uncover<5->{The right column is $\partial_{\theta_j}\mathcal{R}_{\text{reg},\thetav_{\neq j}}$ --- the optimality input for (a)(v).}
\end{frame}

\begin{frame}{Exercise 1 (a) (v)}

	\textbf{Question (a) (v)}: For a real-valued convex function $f$, the global minimum (if it exists) can be characterized in the following way:
		
		\begin{center}
		\setlength{\fboxsep}{8pt}%
		\colorbox{blue!10}{%
		\begin{minipage}{0.9\textwidth}
		\centering
		A point $\theta^* \in \mathbb{R}$ is the global minimum of $f$ if and only if $0 \in \partial f(\theta^*)$.
		\end{minipage}}
	\end{center} 
		
		With this show that 
		$$\theta_j^* \in \argmin_{\theta_j \in \mathbb{R}} \mathcal{R}_{\text{reg}, \thetav_{\neq j}} \Leftrightarrow \theta_j^* = 
		\begin{cases}
			\frac{\rho_j + \lambda}{z_j} \qquad & \text{for } \rho_j < - \lambda \\
			0 \qquad & \text{for } - \lambda \leq \rho_j \leq \lambda \\
			\frac{\rho_j - \lambda}{z_j} \qquad & \text{for } \rho_j > \lambda  
		\end{cases}
		$$
\end{frame}

\begin{frame}{Exercise 1 (a) (v): Solve $0 \in \partial_{\theta_j}\mathcal{R}_{\text{reg}}$ in each sign-case}
	\small
	\textbf{Solution}: set $0 \in \partial_{\theta_j}\mathcal{R}_{\text{reg},\thetav_{\neq j}}$, the subgradient from (a)(iv) (with $z_j=\|\Xmat_{\cdot j}\|_2^2>0$):
	\[ \partial_{\theta_j}\mathcal{R}_{\text{reg},\thetav_{\neq j}} = \begin{cases}
		\{-\rho_j+\theta_j z_j-\lambda\} & \theta_j<0, \\
		[-\rho_j-\lambda,\ -\rho_j+\lambda] & \theta_j=0, \\
		\{-\rho_j+\theta_j z_j+\lambda\} & \theta_j>0.
	\end{cases} \]
	Solve each case for $\theta_j^*$. For the singletons, $0\in\{x\}$ means $0=x$; the assumed sign then forces a condition on $\rho_j$:
	\begin{itemize}
		\item<2-> $\theta_j<0$: $\;0 = -\rho_j+\theta_j z_j-\lambda \Rightarrow \theta_j^*=\tfrac{\rho_j+\lambda}{z_j}$; with $z_j>0$, $\;\theta_j^*<0 \Rightarrow$ \colorbox{blue!10}{$\rho_j<-\lambda$}.
		\item<3-> $\theta_j=0$: $\;0\in[-\rho_j-\lambda,\,-\rho_j+\lambda] \Rightarrow -\rho_j-\lambda\le0\le-\rho_j+\lambda \Rightarrow$ \colorbox{blue!10}{$-\lambda\le\rho_j\le\lambda$}.
		\item<4-> $\theta_j>0$: $\;0 = -\rho_j+\theta_j z_j+\lambda \Rightarrow \theta_j^*=\tfrac{\rho_j-\lambda}{z_j}$; with $z_j>0$, $\;\theta_j^*>0 \Rightarrow$ \colorbox{blue!10}{$\rho_j>\lambda$}.
	\end{itemize}
	
\end{frame}

\begin{frame}{Exercise 1 (a) (v): Continued}
	Summarize everthing up:
	$$\theta_j^* \in \argmin_{\theta_j \in \mathbb{R}} \mathcal{R}_{\text{reg}, \thetav_{\neq j}} \Leftrightarrow \theta_j^* = 
		\begin{cases}
			\frac{\rho_j + \lambda}{z_j} \qquad & \text{for } \rho_j < - \lambda \\
			0 \qquad & \text{for } - \lambda \leq \rho_j \leq \lambda \\
			\frac{\rho_j - \lambda}{z_j} \qquad & \text{for } \rho_j > \lambda  
		\end{cases}
	$$
\end{frame}

\begin{frame}{Exercise 1 (b)}
	\textbf{Question (b)}: for non-singular $\Xmat^\top\Xmat$, find $\Amat \in \mathbb{R}^{p\times p}$ with
	$$\Amat^\top \Xmat^\top\Xmat\, \Amat = \Imat.$$
	\emph{Hint: $\Xmat^\top\Xmat$ is positive definite.}
	\medskip

	\uncover<2->{\textbf{Use the hint:} what structure does positive-definiteness give us?}
	\medskip

	\uncover<3->{%
	\begin{center}
		\setlength{\fboxsep}{8pt}%
		\colorbox{blue!10}{%
		\begin{minipage}{0.92\textwidth}
		\centering
		Recall --- a positive definite matrix has a spectral decomposition
		\[ \Xmat^\top\Xmat = \Vmat\Dmat\Vmat^\top, \quad \Vmat^\top\Vmat = \Imat, \quad \Dmat = \mathrm{diag}(d_1,\dots,d_p),\ d_i > 0. \]
		\end{minipage}}
	\end{center}}
\end{frame}

\begin{frame}{Exercise 1 (b): solve for $\Amat$}
	Want $\Amat^\top \Xmat^\top\Xmat\, \Amat = \Imat$, using $\Xmat^\top\Xmat = \Vmat\Dmat\Vmat^\top$.
	\medskip
	\begin{itemize}
		\item<2-> plug in: $\;\Amat^\top \Vmat\Dmat\Vmat^\top \Amat = \Imat$
		\item<3-> substitute $\Pmat := \Vmat^\top\Amat$: $\;\Pmat^\top\Dmat\Pmat = \Imat$
		\item<4-> $\Dmat$ diagonal, $d_i>0 \Rightarrow \Pmat = \Dmat^{-0.5}$ \quad (then $\Dmat^{-0.5}\Dmat\,\Dmat^{-0.5}=\Imat$)
		\item<5-> undo the substitution: $\Vmat^\top\Amat = \Dmat^{-0.5}$, $\Vmat$ orthogonal $\Rightarrow$ \colorbox{blue!10}{$\Amat = \Vmat\Dmat^{-0.5}$}
	\end{itemize}
\end{frame}

\begin{frame}{Exercise 1 (c) (i)}
	(c) For a design matrix with orthornormal columns, i.e., $\Xmat^T \Xmat = \Imat$, exists an analytical minimizer of the Lasso regression $\thetavh_{Lasso} = (\thetah_{Lasso, 1}, \ldots, \thetah_{Lasso, p})^T$ that is given by 
	$$\thetah_{Lasso, i} = \sign(\thetah_{i}) \max \left\{ \left|\thetah_i \right|  - \lambda, 0   \right\}, \quad i = 1, \ldots, p,$$
	where $\thetavh = (\thetah_1, \ldots, \thetah_p)^T = (\Xmat^T \Xmat)^{-1} \Xmat^T \yv$ is the minimizer of the unregularized empirical risk (w.r.t. the L2 loss).
	
	Under the assumption that $\Xmat^T\Xmat$ is non-singular, your colleague proposes to project $\Xmat$ with $\Amat$ from (b), i.e., use $\tilde{\Xmat} = \Xmat \Amat$ and then apply the analytical solution given here. 
	\vspace{10pt}
	
	\textbf{(i)} Show that this does not generally solve the original Lasso regression.
	 
	\vspace{10pt} 
	\emph{Hint: You only need to check under which condition $\nabla_{\thetav} 0.5 || \Xmat \thetav - \yv||_2^2 = \nabla_{\thetav} 0.5 ||\Xmat \Amat \thetav - \yv||_2^2 $. The proof can be finished with a subgradient argument regarding stationarity, which you do not have to do.}
\end{frame}

\begin{frame}{Exercise 1 (c) (i): Continued}
	\textbf{Solution}: compare the two gradients (the hint).
	\begin{align*}
		\onslide<2->{\nabla_{\thetav}\, 0.5\|\Xmat\Amat\thetav - \yv\|_2^2 &= \Amat^\top\Xmat^\top\Xmat\Amat\,\thetav - \Amat^\top\Xmat^\top\yv} \\
		\onslide<3->{\nabla_{\thetav}\, 0.5\|\Xmat\thetav - \yv\|_2^2 &= \Xmat^\top\Xmat\,\thetav - \Xmat^\top\yv}
	\end{align*}
	\uncover<4->{equal $\Rightarrow \Amat^\top\Xmat^\top\Xmat\Amat = \Xmat^\top\Xmat \overset{(b)}{\Longrightarrow} \Xmat^\top\Xmat = \Imat$, \;i.e.\ \colorbox{blue!10}{$\Amat = \Imat$}.}
	\medskip
	\uncover<5->{$\Amat \neq \Imat \;\Rightarrow\;$ projected solution $\neq$ original Lasso.}
\end{frame}


\begin{frame}{Exercise 1 (c) (ii)}
	\textbf{Question (c) (ii)}: How to adapt the penalty so the projected problem matches the original Lasso? Can we still solve coordinate-wise?
	\medskip

	\textbf{Solution}:
	\begin{itemize}
		\item<2-> penalize $||\Amat\thetav||_1$ (not $||\thetav||_1$) $\Rightarrow \thetav^*_{\text{proj}} = \Amat^{-1}\thetav_{\text{Lasso}}$ --- same minimal risk, \colorbox{blue!10}{equivalent}.
		\item<3-> but $||\Amat\thetav||_1$ couples the coordinates $\Rightarrow$ no per-coordinate split --- not solvable independently.
	\end{itemize}
	\medskip

	\uncover<3->{%
	\begin{center}
	\begin{tikzpicture}[scale=0.78]

		\node at (0,-2.3) {\scriptsize separable, sparse corners};
		\begin{scope}[xshift=5.6cm]
			\draw[->,gray!55] (-1.7,0)--(1.7,0) node[right,black]{\scriptsize$\theta_1$};
			\draw[->,gray!55] (0,-1.3)--(0,1.6) node[above,black]{\scriptsize$\theta_2$};
			\draw[thick,fill=orange!18,rotate=35,xscale=1.35,yscale=0.72] (1,0)--(0,1)--(-1,0)--(0,-1)--cycle;
			\node at (0,-1.85) {\scriptsize$\|\Amat\thetav\|_1\le t$: tilted};
			\node at (0,-2.3) {\scriptsize coupled, corners off-axis};
		\end{scope}
	\end{tikzpicture}
	\end{center}}
\end{frame}

\begin{frame}{Exercise 1 (c) (iii)}
	\textbf{Question (c) (iii)}: Does the procedure in (c) perform variable selection?
	\medskip

	\textbf{Solution}:
	\begin{itemize}
		\item<2-> $||\tilde{\Xmat}\thetav - \yv||_2^2 + \lambda||\thetav||_1$ with $\tilde{\Xmat}=\Xmat\Amat$ --- Lasso in the \emph{projected} coordinates.
		\item<3-> so $\thetav^*$ is sparse on $\tilde{\Xmat}$ --- selects \emph{projected} features.
		\item<4-> but $\Xmat\Amat\thetav^* = \Xmat(\Amat\thetav^*)$, and $\Amat\thetav^*$ is generally dense $\Rightarrow$ \colorbox{blue!10}{no selection on $\Xmat$}.
	\end{itemize}
	\medskip

	\uncover<4->{%
	\begin{center}
	\begin{tikzpicture}[scale=0.55]
		\draw[gray!50] (-0.3,0)--(5.3,0);
		\draw[very thick,blue!70!black] (2,0)--(2,1.6); \fill[blue!70!black] (2,1.6) circle (2.4pt);

\begin{frame}{Exercise 1 (d)}
	(d) You are given the following code to compare the quality of the projected Lasso regression vs.\ the regular Lasso regression. Complete the missing code of the algorithms and interpret the result.
	\medskip

	The two estimators are ported from the R reference solution to Python. Part (b) gives $\Amat = \Vmat \Dmat^{-0.5}$, which orthonormalizes the design so the closed-form Lasso applies; the regular Lasso uses the coordinate-descent update derived in (a)(v).
\end{frame}

\begin{frame}[fragile]{Exercise 1 (d): the two estimators (Python)}
	\begin{columns}[T]
	\begin{column}{0.5\textwidth}
\begin{lstlisting}[style=pystyle, basicstyle=\ttfamily\tiny]
def proj_orth_lasso(X, y, lam):
    vals, vecs = np.linalg.eigh(X.T @ X)
    A = vecs @ np.diag(vals ** -0.5)
    Xt = X @ A                 # Xt'Xt = I
    b = Xt.T @ y               # OLS, orthonormal
    b = np.sign(b) * np.maximum(
        np.abs(b) - lam, 0.0)  # soft-threshold
    return A @ b               # back to X-coords
\end{lstlisting}
	\end{column}
	\begin{column}{0.5\textwidth}
\begin{lstlisting}[style=pystyle, basicstyle=\ttfamily\tiny]
def lasso(X, y, lam, N):
    p = X.shape[1]
    theta = np.ones(p)
    for i in range(1, N + 1):
        j = i % p
        mask = np.ones(p); mask[j] = 0.0
        rho = X[:, j] @ (y - X @ (theta * mask))
        z = np.sum(X[:, j] ** 2)
        if rho < -lam:   theta[j] = (rho+lam)/z
        elif rho > lam:  theta[j] = (rho-lam)/z
        else:            theta[j] = 0.0
    return theta
\end{lstlisting}
	\end{column}
	\end{columns}
	\medskip
	\footnotesize Left: project $\to$ closed-form soft-threshold $\to$ map back via $\Amat$. Right: cyclic coordinate descent applying the $(a)(v)$ update to one $\theta_j$ at a time.
\end{frame}

		\textbf{\textcolor{orange!75!black}{Take-aways}}
		\begin{itemize}\setlength{\itemsep}{1pt}
			\item no general closed form $\Rightarrow$ coordinate descent; the per-coordinate step is \emph{soft-thresholding}.
			\item orthonormal design ($\Xmat^\top\Xmat = \Imat$) is the special case with an exact soft-threshold formula.
			\item L1 selects only in the basis it penalizes; rotating the design leaves $\thetavh$ dense in the original coordinates.
