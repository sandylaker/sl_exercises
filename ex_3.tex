\begin{frame}{Exercise 3: Connection between between MLE and ERM}
\small
	Suppose we are facing a regression task, i.e., $\Yspace = \mathbb{R}$, and the feature space $\Xspace \subseteq \mathbb{R}^p$. Let us assume that the relationship between the features and labels is specified by 
	$$y = m^{-1}(m(\ftrue(\xv)) + \eps),$$
	where $m: \mathbb{R} \to \mathbb{R}$ is a continuous strictly monotone function with $m^{-1}$ being its inverse function, and the errors are Gaussian, i.e., $\eps \sim \mathcal{N}(0, \sigma^2)$. In particular, for the data points $(\xv^{(1)}, y^{(1)}, \ldots, \xv^{(n)}, y^{(n)})$ it holds that
	$$\yi = m^{-1}(m(\ftrue(\xv^{(i)})) + \epsi),$$
	where $\eps^{(1)}, \ldots, \eps^{(n)}$ are i.i.d. with distribution $\mathcal{N}(0, \sigma^2)$.
	
	\textbf{Disclaimer:} We assume in the following that $\my$ and $\mfx$ is well-defined for any $y \in \Yspace, f \in \Hspace$ and $\xv \in \Xspace$.
	
	(a) How can we transform the labels $y^{(1)}, \ldots, y^{(n)}$ to ``new'' labels $z^{(1)}, \ldots, z^{(n)}$ such that $ z^{(i)} \ | \ \xv$ is normally distributed? What are the parameters of this normal distribution?
	
\end{frame}

\begin{frame}{Solution: Question (a)}
\small
	\textcolor{gray}{\footnotesize Goal: transform $\yi \rightsquigarrow \zi$ such that $\zi \mid \xv^{(i)}$ is Gaussian.}
	\vspace{10pt}

	\begin{itemize}\setlength{\itemsep}{6pt}
		\item<1-> Apply $m$:\quad $\yi = m^{-1}(\mfruexi + \epsi) \;\Longrightarrow\; \myi = \mfruexi + \epsi.$
		\item<2-> Set\quad $\zi := \myi.$
		\item<3-> Condition on $\xv^{(i)}$:\quad $\mfruexi$ fixed,\; $\epsi \sim \mathcal{N}(0, \sigma^2)$\; $\Rightarrow$\; $\zi \mid \xv^{(i)}$ shifted Gaussian.
		\item<4-> $\Longrightarrow\; \zi \mid \xv^{(i)} \sim \mathcal{N}(\mfruexi,\, \sigma^2).$
	\end{itemize}
\end{frame}

\begin{frame}{Question (b)}
\small
\begin{columns}[T]
\begin{column}{0.55\textwidth}
	Assume that the hypothesis space is
	\vspace{2pt}
	\noindent\resizebox{\linewidth}{!}{$
		\begin{aligned}
			\Hspace = \{ f(\cdot, \thetav): \Xspace \to \mathbb{R} \mid f(\cdot | \thetav) & \text{ belongs to a certain functional} \\[-2pt]
			& \text{family parameterized by } \thetav \in \Theta \},
		\end{aligned}
	$}
	\vspace{2pt}
	where $\thetav = (\theta_1, \theta_2, \ldots, \theta_d))$ is a parameter vector, which is an element of a \textbf{parameter space} $\Theta$. Based on your findings (a), establish a relationship between minimizing the negative log-likelihood for $(\xv^{(1)}, z^{(1)}),
	\ldots (\xv^{(n)}, z^{(n)})$ and empirical loss minimizing over $\Hspace$ of the generalized L2-loss function of Exercise sheet 1, i.e., $L(y, \fx)) = (\my - \mfx)^2$.
\end{column}
\begin{column}{0.43\textwidth}
	\textcolor{gray}{Roadmap --- two recipes to connect:}
	\vspace{6pt}

	\textbf{\textcolor{blue!60!black}{MLE on $(\xi, \zi)$.}}
	\begin{enumerate}\setlength{\itemsep}{3pt}
		\item<1-> Write the likelihood $\mathcal{L}(\thetav) = {\textstyle\prod_{i=1}^n} p(\zi \mid \xi, \thetav)$.
		\item<2-> Take $-\log$ $\to$ NLL $-\ell(\thetav)$.
	\end{enumerate}

	\vspace{8pt}
	\textbf{\textcolor{red!60!black}{ERM with generalized L2.}}
	\begin{enumerate}\setlength{\itemsep}{3pt}
		\item<3-> Empirical risk $\riske(f) = {\textstyle\sum_{i=1}^n} L(\yi, f(\xi))$.
		\item<4-> Plug in $L(y, \fx) = (\my - \mfx)^2$.
	\end{enumerate}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Solution to Question (b)}
\small
\begin{columns}[T]
\begin{column}{0.64\textwidth}
	The likelihood for $(\xv^{(1)}, z^{(1)}), \ldots, (\xv^{(n)}, z^{(n)})$:
	\begin{align*}
		\only<1->{
			\mathcal{L}(\thetav)
			&= {\textstyle\prod_{i=1}^n} p\!\left( \zi \mid f(\xi \mid \thetav),\, \sigma^2 \right)  \\}
		\only<2->{&\propto \exp\!\left( -\tfrac{1}{2\sigma^2} {\textstyle\sum_{i=1}^n} \left[ \zi - m(f(\xi \mid \thetav)) \right]^2 \right).}
	\end{align*}
	\uncover<3->{Take $-\log$ $\to$ NLL:}
	\begin{align*}
		\only<4->{- \ell(\thetav) &= - \log \mathcal{L}(\thetav) \\}
		\only<5->{&\propto {\textstyle\sum_{i=1}^n} \left[ \zi - m(f(\xi \mid \thetav)) \right]^2 \\}
		\only<6->{&= {\textstyle\sum_{i=1}^n} \left[ \myi - m(f(\xi \mid \thetav)) \right]^2.}
	\end{align*}

	\uncover<7->{\textbf{$\Rightarrow$} NLL $\propto$ empirical risk under generalized L2-loss.}
\end{column}
\begin{column}{0.34\textwidth}
	\textbf{\textcolor{blue!60!black}{MLE.}}
	\begin{enumerate}\setlength{\itemsep}{2pt}
		\item Write likelihood\only<2->{\;\textcolor{green!50!black}{$\checkmark$}}
		\item Take $-\log$ $\to$ NLL\only<4->{\;\textcolor{green!50!black}{$\checkmark$}}
	\end{enumerate}

	\vspace{6pt}
	\textbf{\textcolor{red!60!black}{ERM target.}}
	\begin{enumerate}\setlength{\itemsep}{2pt}
		\item Empirical risk\\ ${\textstyle\sum_{i=1}^n} L(\yi, f(\xi))$
		\item Plug in $(\my - \mfx)^2$\only<6->{\;\textcolor{green!50!black}{$\checkmark$}}
	\end{enumerate}

	\vspace{6pt}
	\uncover<7->{
		\centering
		\textcolor{orange!75!black}{$\Rightarrow$ NLL $\equiv$ ERM (up to const.).}
	}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Question (c)}
	(c) In many practical applications, one often observed statisical property is that the label $y$ given $\xv$ follows a \emph{log-normal distribution}. Note that we can obtain such a relationship by using $m(x) = \log(x)$ above. In the following we want to consider the conjecture of James D. Forbes, who conjectured in the year 1857 that the relationship between the air pressure $y$ and the boiling point of water $x$ is given by $$y = \theta_1 \exp(\theta_2 x + \eps),$$ for some specific values $\theta_1 \in \mathbb{R}_+, \theta_2 \in \mathbb{R}$ and some error terms $\eps$ (of course, we assume that this error term is stochastic and normally distributed).
	
	What would be a suitable hypothesis space $\Hspace$ if this conjecture holds?
\end{frame}

\begin{frame}{Solution to Question (c): setup \& decomposition}
	\small
	\begin{itemize}\setlength{\itemsep}{5pt}
		\item<1-> \textbf{Framework recap:}\; $y = m^{-1}(m(f(\xv)) + \eps)$,\; $\eps \sim \mathcal{N}(0, \sigma^2)$.\;
			Two ingredients:
			\begin{itemize}\setlength{\itemsep}{0pt}
				\item[$\circ$] $f$ --- deterministic regression function,\; $f \in \Hspace$ (what we choose);
				\item[$\circ$] $m$ --- fixed link function (\emph{not} part of $\Hspace$);
				\item[$\circ$] $\eps$ --- Gaussian noise, applied \emph{after} $m$.
			\end{itemize}
		\item<2-> \textbf{What Q(c) is really asking:}\; pick the form of $f(\cdot \mid \thetav)$ so that the framework reproduces Forbes' $y = \theta_1 \exp(\theta_2 x + \eps)$.
		\item<3-> \textbf{Why $m = \log$?}\; Q(c) says $y \mid \xv$ log-normal $\Longleftrightarrow$ $\log y \mid \xv$ Gaussian.\;
			Framework demands Gaussian residual after $m$\; $\Longrightarrow$\; $m = \log$,\; $m^{-1} = \exp$.
		\item<4-> \textbf{Decompose:}\; with $m = \log$ the framework simplifies to
			\[
				y = \exp(\log f(\xv) + \eps) = f(\xv) \cdot \exp(\eps),
			\]
			\textcolor{gray}{\footnotesize i.e.\ deterministic signal $\times$ multiplicative noise. Match Forbes' to this shape next.}
	\end{itemize}
\end{frame}

\begin{frame}{Solution to Question (c): algebra}
	\small
	\textcolor{gray}{\footnotesize Target shape (from setup):\; $y = f(\xv) \cdot \exp(\eps)$.}
	\vspace{6pt}
	\begin{itemize}\setlength{\itemsep}{5pt}
		\item<1-> \textbf{Reshape Forbes' into the target shape:}
			\begin{align*}
				y &= \theta_1 \exp(\theta_2 x + \eps) \\
				\only<2->{&= \theta_1 \exp(\theta_2 x) \cdot \exp(\eps) \\}
				\only<3->{&= \underbrace{\theta_1 \exp(\theta_2 x)}_{=\, f(x \mid \thetav)} \cdot \exp(\eps).}
			\end{align*}
		\item<4-> \textbf{Read off $f$ \& conclude:}
			$$\Hspace = \{ f(x \mid \thetav) = \theta_1 \exp(\theta_2 x) : \thetav \in \Theta \}.$$
		\item<5-> \textbf{Sanity check (reparam.):}\; $\beta_0 := \log\theta_1$,\; $\beta_1 := \theta_2$\;
			$\Longrightarrow$\; $\log y = \beta_0 + \beta_1 x + \eps$\;
			\textcolor{gray}{\footnotesize (linear Gaussian regression on $\log y$)}.
	\end{itemize}
\end{frame}

\begin{frame}{Solution to Question (c): alternative --- linear in $\log y$}
	\small
	\begin{itemize}\setlength{\itemsep}{4pt}
		\item<1-> \textbf{Apply $\log$ to both sides of Forbes':}
			\[
				y = \theta_1 \exp(\theta_2 x + \eps)
				\;\Longleftrightarrow\;
				\log y = \log\theta_1 + \theta_2 x + \eps.
			\]
			\textcolor{gray}{\footnotesize $\Longrightarrow$ ordinary linear regression on $z := \log y$.}
		\item<2-> \textbf{Linear hypothesis space} (with $\xv = (1, x)^\top \in \Xspace = \{1\}\times\mathbb{R}$):
			\[
				\Hspace = \{ f(\xv \mid \thetav) = \log(\theta_1)\, x_1 + \theta_2\, x_2 : \thetav \in \Theta\}
				= \{\, \xv^\top \thetav \,\}.
			\]
		\item<3-> \textbf{ERM = OLS:}\;
			$\hat{\thetav} = (\log\hat\theta_1,\, \hat\theta_2)^\top = (\Xmat^\top \Xmat)^{-1} \Xmat^\top \bm{z}$,\;\;
			$\bm{z} = (\log y^{(1)}, \ldots, \log y^{(n)})^\top$.
		\item<4-> \textbf{Closed form (simple-regression case):}
			\[
				\hat\theta_2 = \frac{\sum_i (\xi_2 - \bar{x}_2)(\log\yi - \overline{\log y})}{\sum_i (\xi_2 - \bar{x}_2)^2},
				\quad
				\hat\theta_1 = \exp\!\bigl(\overline{\log y} - \hat\theta_2\, \bar{x}_2\bigr).
			\]
	\end{itemize}
\end{frame}

\begin{frame}[fragile]{Solution to Question (c): R code \& fit on Forbes' data}
\scriptsize
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\begin{lstlisting}[style=Rstyle]
library(MASS); data(forbes); attach(forbes)
X <- cbind(rep(1, 17), bp); y <- pres

# (1) numeric ERM on Forbes' form
f <- function(x, theta)
  exp(theta[2]*x[2]) * theta[1] * x[1]
emp_risk <- function(theta)
  sum((log(y) - log(apply(X, 1, f, theta)))^2)
hat_theta <- optim(c(0.4, 0.5), emp_risk,
                   method = "L-BFGS-B",
                   lower = c(0, -Inf),
                   upper = c(Inf, Inf))$par
print(hat_theta)
## [1] 0.38050940 0.02059961

# (2) closed-form OLS on log(y)
hat_theta_2 <- cov(bp, log(pres)) / var(bp)
hat_theta_1 <- exp(mean(log(pres))
                   - hat_theta_2 * mean(bp))
\end{lstlisting}

\begin{frame}{Recap: link to the lecture material}
\scriptsize
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{c|p{6.2cm}|p{6.2cm}}
\textbf{Q} & \textbf{What the exercise teaches} & \textbf{Lecture material it builds on} \\
\hline
(a) &
The substitution $\zi = m(\yi)$ exposes the hidden Gaussian:\; $\zi \mid \xv^{(i)} \sim \mathcal{N}(m(\ftrue(\xv^{(i)})), \sigma^2)$. The transformed labels obey the lecture's standard regression form. &
Standard regression DGP $y = \ftrue(\xv) + \eps$ with $\eps \sim \mathcal{N}(0, \sigma^2)$. The lecture only treats the identity link $m = \mathrm{id}$, where the Gaussian noise lives directly on the $y$-scale. \\
(b) &
Define $\text{loss} := -\log p$. Then minimizing NLL \emph{is} empirical-risk minimization, so MLE and ERM are the same optimization. Plugging in the Gaussian density on $\zi$ produces the generalized L2 loss $L(y, f(\xv)) = (m(y) - m(f(\xv)))^2$. &
The lecture's loss-as-negative-log-likelihood identity that fixes a loss for every noise model:\; Gaussian $\to$ L2,\; Laplace $\to$ L1,\; Bernoulli $\to$ log-loss. The exercise extends this catalogue with log-normal $\to$ generalized L2. \\
(c) &
Choosing $\Hspace$ $=$ choosing the functional form of $f(\cdot \mid \thetav)$. The hint $y \mid \xv$ log-normal pins $m = \log$;\; matching Forbes' to the framework yields $\Hspace = \{\theta_1 \exp(\theta_2 x) : \thetav \in \Theta\}$, fittable as OLS on $\log y$. &
Hypothesis space as a parametric family $\{f_{\thetav} : \thetav \in \Theta\}$;\; convex losses for exponential-family / GLM models. The derivation follows the same MLE-to-loss pattern as the log-cosh $\leftrightarrow$ cosh-MLE example shown in the lecture. \\
\end{tabular}
\end{frame}
