\begin{frame}{Exercise 1: Bayesian Linear Model}
\small
\textbf{Model:}
\begin{equation*}
	y = \fx + \eps = \thetav^\top \xv + \eps,
\end{equation*}
with $\eps \sim \Norm(0, \sigma^2) \perp \xv$. Per data point:
\begin{equation*}
	\yi = \thetav^\top \xi + \epsi, \qquad \epsi \sim \Norm(0, \sigma^2) \text{ i.i.d.}, \ i \in [n].
\end{equation*}
\textbf{Bayesian view:} $\thetav$ is random with a prior.

\smallskip
\textbf{MAP estimate:} $\thetavh = \argmax_{\thetav} p(\thetav | \Xmat, \yv)$.

\smallskip
\textbf{(a)} Uniform prior $q(\thetav) \propto 1$ $\Rightarrow$ show MAP $=$ \textbf{L2-loss ERM} (OLS) over linear models.
\end{frame}

\begin{frame}{Exercise 1: The Key Insight --- MAP = Regularized ERM}
	\small
	Take $-\log$ of the posterior: maximizing $p(\thetav|\Xmat,\yv)$ becomes minimizing
	\begin{equation*}
		\thetavh = \argmax_{\thetav} p(\thetav | \Xmat, \yv)
		= \argmin_{\thetav} \Big[ \underbrace{-\log p(\yv | \Xmat, \thetav)}_{\text{data fit (L2 loss)}} + \underbrace{(-\log q(\thetav))}_{\text{penalty}} \Big].
	\end{equation*}
	\begin{itemize}
		\item The likelihood is always Gaussian $\leadsto$ data-fit term is the \textbf{L2 loss}.
		\item The \textbf{prior choice $q(\thetav)$ determines the penalty}. Parts (a)/(b)/(c) just swap the prior:
	\end{itemize}
	\medskip
	\begin{center}
	\renewcommand{\arraystretch}{1.3}
	\begin{tabular}{@{}llll@{}}
		\toprule
		\textbf{Prior $q(\thetav)$} & \textbf{$-\log q(\thetav)$ (penalty)} & \textbf{Estimator} & \textbf{$\lambda$} \\
		\midrule
		(a) Uniform & const (none) & OLS / L2-loss ERM & --- \\
		(b) Gaussian & $\tfrac{1}{2\tau^2}\|\thetav\|_2^2$ & Ridge & $\sigma^2/\tau^2$ \\
		(c) Laplace & $\tfrac{1}{\tau}\|\thetav\|_1$ & Lasso & $2\sigma^2/\tau$ \\
		\bottomrule
	\end{tabular}
	\end{center}
\end{frame}

\begin{frame}{Exercise 1: Why the Prior Is the Regularizer}
	\small
	The penalty $-\log q(\theta)$ as a function of a scalar parameter $\theta$ --- its \emph{shape} is what each prior contributes:
	\medskip
	\begin{center}
	\begin{tikzpicture}[font=\scriptsize, >=Latex, scale=1.0]
		% axes
		\draw[->] (-2.6,0) -- (2.8,0) node[right] {$\theta$};
		\draw[->] (0,-0.3) -- (0,2.5) node[above] {$-\log q(\theta)$};
		% uniform: flat (no penalty)
		\draw[very thick, black!55] (-2.4,0.25) -- (2.4,0.25);
		\node[black!55, anchor=west] at (0.55,0.85) {uniform (flat)};
		% gaussian: parabola (L2)
		\draw[very thick, blue!65!black, domain=-1.55:1.55, samples=60, smooth] plot (\x, {0.85*\x*\x});
		\node[blue!65!black, anchor=east] at (-1.35,2.15) {Gaussian: $\propto\theta^2$ (L2)};
		% laplace: V-shape (L1)
		\draw[very thick, red!70!black] (-2.0,2.0) -- (0,0) -- (2.0,2.0);
		\node[red!70!black, anchor=west] at (1.05,2.05) {Laplace: $\propto|\theta|$ (L1)};
	\end{tikzpicture}
	\end{center}
	\medskip
	\begin{itemize}
		\item \textbf{Flat} $\Rightarrow$ no shrinkage (OLS). \quad \textbf{Parabola} $\Rightarrow$ smooth shrinkage (Ridge).
		\item \textbf{V-shape} has a kink at $0$ $\Rightarrow$ drives coefficients exactly to $0$ (Lasso, sparsity).
	\end{itemize}
\end{frame}

\begin{frame}{Exercise 1: Think Before Solving}
	\small
	\textbf{Target:}\quad turn the $\argmax$ of the posterior into a familiar $\argmin$ (an ERM).
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{From max to min.} $\log$ is monotone --- what does $-\log p(\thetav|\Xmat,\yv)$ do to the $\argmax$? How does Bayes split it into likelihood $\times$ prior?
		\item<2-> \textbf{Data fit.} The likelihood $\yv|\Xmat,\thetav$ is Gaussian. Which familiar loss does $-\log p(\yv|\Xmat,\thetav)$ reduce to?
		\item<3-> \textbf{Penalty.} For each prior (uniform / Gaussian / Laplace), what does $-\log q(\thetav)$ become, and why does the marginal $p(\yv|\Xmat)$ never matter?
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{$\argmax p(\thetav|\Xmat,\yv) \to \argmin[\,-\log\text{lik.} - \log\text{prior}\,]$ = (L2 loss) $+$ (penalty)}}
\end{frame}

\begin{frame}{1 (a): Construct Posterior from Bayes' Rule}
	\begin{equation*}
		\begin{aligned}
			\only<1->{
			\underbrace{p(\thetav | \Xmat, \yv)}_{\text{posterior}} &= \frac{p(\thetav, \yv| \Xmat)}{p(\yv | \Xmat)} \\
			}
			\only<2->{
			&= \frac{\overbrace{p(\yv | \Xmat, \thetav)}^{\text{likelihood}} \overbrace{q(\thetav)}^{\text{prior}} }{\underbrace{p(\yv | \Xmat)}_{\text{marginal}}}
			}
		\end{aligned}
	\end{equation*}
	
	\begin{itemize}
			\item<3-> Linear model: $\yv | \Xmat, \thetav \sim \Norm(\Xmat \thetav, \sigma^2 \Imat)$ (next slide).
			\item<4-> \textbf{Uniform prior:} $q(\thetav) \propto 1$.
			\item<5-> Marginal $p(\yv | \Xmat) \perp \thetav$ $\leadsto$ constant for the $\argmax$.
	\end{itemize}
\end{frame}

\begin{frame}[t]{1 (a): The Likelihood $p(\yv | \Xmat, \thetav)$}
	\begin{equation*}
		\begin{aligned}
			p(\yv | \Xmat, \thetav) &\propto \exp \left[ - \frac{1}{2\sigma^2} (\yv - \Xmat\thetav)^\top (\yv - \Xmat \thetav) \right] \\
			&= \exp \left[- \frac{||\yv - \Xmat \thetav||^2}{2 \sigma^2}\right] \\
			&= \exp \left[ -\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2}  \right].
		\end{aligned}
	\end{equation*}
	\uncover<2->{

	Recall: prior $q(\thetav) \propto 1$; marginal $p(\yv | \Xmat)$ irrelevant.

	$\leadsto p(\thetav | \Xmat, \yv) \propto p(\yv | \Xmat, \thetav)\, q(\thetav)$.
	}
\end{frame}

\begin{frame}{1 (a): MAP Estimate}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	\footnotesize
	\begin{equation*}
			p(\thetav | \Xmat, \yv) \propto p(\yv | \Xmat, \thetav) \cdot q(\thetav)
			\propto \exp \left[ -\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2}  \right].
	\end{equation*}
	\uncover<2->{
	\setlength{\jot}{2pt}
	\begin{equation*}
		\begin{aligned}
			\thetavh &= \argmax_{\thetav} p(\thetav | \Xmat, \yv)
				= \argmax_{\thetav} \log p(\thetav | \Xmat, \yv) \\
				&= \argmax_{\thetav}  -\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2}
				= \argmin_{\thetav} \frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} \\
				&= \argmin_{\thetav} \sum_{i=1}^n (\yi - \thetav^\top \xi)^2. \hfill\qed
		\end{aligned}
	\end{equation*}
$\Rightarrow$ MAP $\Leftrightarrow$ L2-loss ERM (OLS).
}
	\end{column}
	\begin{column}{0.24\textwidth}
	\rmmap{1}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1(b)}
	\textbf{Gaussian prior:}
	\begin{equation*}
		q(\thetav) \propto \exp \left[ - \frac{1}{2 \tau^2} \thetav^\top \thetav \right], \quad \tau > 0,
	\end{equation*}
	Show: for a suitable $\tau$, MAP $=$ \textbf{Ridge} (L2-loss $+$ L2 penalty) over linear models.
\end{frame}

\begin{frame}{1 (b): Posterior with a Gaussian Prior}
\begin{equation*}
	\begin{aligned}
		p(\thetav | \Xmat, \yv) &\propto p(\yv | \Xmat, \thetav) q(\thetav) \\
		&\propto \exp \left[-\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2}- \frac{1}{2 \tau^2} \thetav^\top \thetav \right] \\
		&\propto \exp \left[-\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} - \frac{||\thetav||_2^2}{2 \tau^2} \right].
	\end{aligned}
\end{equation*}
Next: $\argmax_{\thetav} p(\thetav | \Xmat, \yv) = \argmax_{\thetav} \log p(\thetav | \Xmat, \yv)$.
\end{frame}

\begin{frame}{1 (b): MAP Estimate}
\small
\begin{columns}[T]
\begin{column}{0.74\textwidth}
\begin{equation*}
	\begin{aligned}
		\thetavh &= \argmax_{\thetav} \log p(\thetav | \Xmat, \yv) \\
		&= \argmax_{\thetav} -\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} - \frac{||\thetav||_2^2}{2 \tau^2} \\
		&= \argmin_{\thetav} \frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} + \frac{||\thetav||_2^2}{2 \tau^2} \\
		&= \argmin_{\thetav} \sum_{i=1}^n(\yi - \thetav^\top \xi)^2 + \frac{\sigma^2}{\tau^2} ||\thetav||_2^2. \hfill\qed
	\end{aligned}
\end{equation*}
{\footnotesize (Because $\times\, 2\sigma^2$ doesn't change the argmin.)}
\medskip

Set $\lambda = \frac{\sigma^2}{\tau^2}$ $\Rightarrow$ MAP $\Leftrightarrow$ L2-loss $+$ L2 penalty (Ridge).
\end{column}
\begin{column}{0.24\textwidth}
\rmmap{2}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Exercise 1(c)}
	\textbf{Laplace prior:}
	\begin{equation*}
		q(\thetav) \propto \exp \left[ - \frac{\sum_{j=1}^{p} |\theta_j|}{\tau} \right], \quad \tau > 0,
	\end{equation*}
	Show: for a suitable $\tau$, MAP $=$ \textbf{Lasso} (L2-loss $+$ L1 penalty) over linear models.
\end{frame}

\begin{frame}[t]{1 (c): Posterior with a Laplace Prior}
	\begin{equation*}
		\begin{aligned}
			p(\thetav | \Xmat, \yv) &\propto p(\yv | \Xmat, \thetav) q(\thetav) \\
			&\propto \exp \left[-\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2}- \frac{||\thetav||_1}{\tau} \right]. \\
		\end{aligned}
	\end{equation*}
\end{frame}

\begin{frame}{1 (c): MAP Estimate}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	\begin{equation*}
		\begin{aligned}
			\thetavh &= \argmax_{\thetav} \log p(\thetav | \Xmat, \yv) \\
			&= \argmax_{\thetav} -\frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} - \frac{||\thetav||_1}{\tau} \\
			&= \argmin_{\thetav} \frac{\sum_{i=1}^n(\yi - \thetav^\top \xi)^2}{2\sigma^2} + \frac{||\thetav||_1}{\tau} \\
			&= \argmin_{\thetav} \sum_{i=1}^n(\yi - \thetav^\top \xi)^2 + \frac{2 \sigma^2}{\tau} ||\thetav||_1. \hfill\qed
		\end{aligned}
	\end{equation*}
	{\footnotesize (Because $\times\, 2\sigma^2$ doesn't change the argmin.)}
	\medskip

	Set $\lambda = \frac{2 \sigma^2}{\tau}$ $\Rightarrow$ MAP $\Leftrightarrow$ L2-loss $+$ L1 penalty (Lasso).
	\end{column}

	\rmmap{3}
	\end{column}

\begin{frame}{2 (a): Toolbox --- Composition Rules}
	\small
	Assume $k_0, k_1$ are valid covariance functions. Proofs below cite these:
	\medskip
	\begin{tcolorbox}[colback=blue!4, colframe=blue!55!black, boxsep=2pt, left=5pt, right=5pt, top=3pt, bottom=3pt, arc=2pt]
	\footnotesize
	\begin{columns}[T]
	\begin{column}{0.49\textwidth}
	\begin{enumerate}
		\setlength{\itemsep}{3pt}
		\item $\kxxp = \xv^\top \xvp$;
		\item $\kxxp = c \cdot \kzxxp$, \ $c \geq 0$ const;
		\item $\kxxp = \kzxxp + \koxxp$;
		\item $\kxxp = \kzxxp \cdot \koxxp$;
	\end{enumerate}
	\end{column}
	\begin{column}{0.49\textwidth}
	\begin{enumerate}
		\setlength{\itemsep}{3pt}
		\setcounter{enumi}{4}
		\item $\kxxp = g(\kzxxp)$, \ $g$ polynomial, \textbf{pos.}\ coeff.;
		\item $\kxxp = t(\xv) \cdot \kzxxp \cdot t(\xvp)$, \ $t$ \textbf{continuous};
		\item $\kxxp = \exp(\koxxp)$;
		\item $\kxxp = \xv^\top \Amat \xvp$, \ $\Amat \succeq 0$.
	\end{enumerate}
	\end{column}
	\end{columns}
	\end{tcolorbox}
\end{frame}

\begin{frame}{2 (a): Think Before Solving}
	\small
	\textbf{Target:}\quad show each covariance function is \emph{valid} (a positive semi-definite kernel).
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{Two routes (single kernel).} (i) Gram matrix $\Kmat$ symmetric with $\vv^\top\Kmat\vv\ge 0\ \forall\vv$, or (ii) $k(\xv,\xvp)=\langle\phi(\xv),\phi(\xvp)\rangle$. Which is easier for the constant kernel?
		\item<2-> \textbf{Build, don't re-derive.} Can you assemble composite kernels from already-valid pieces via the toolbox rules?
		\item<3-> \textbf{Decomposition.} Linear $=$ constant $+$ inner product; polynomial $=(\,\cdot\,)^p$; squared-exp $= t\cdot k_4\cdot t$. Which rule fits each step?
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{PSD Gram matrix \;\;or\;\; feature-map inner product \;\;then\;\; compose via rules 1--8}}
\end{frame}

\begin{frame}{2 (a): Constant Kernel --- Proof via Kernel Matrix}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	Kernel (Gram) matrix $\Kmat \in \mathbb{R}^{n \times n}$ on $\D = \{\xv^{(1)}, \ldots, \xv^{(n)} \}$, $\Kij = \kxxp$.

	Here $\kxxp \equiv \sigma_0^2$, so $\Kij = \sigma_0^2$:
	\begin{equation*}
		\Kmat = \begin{pmatrix}
			\sigma_0^2 & \ldots & \sigma_0^2 \\
			\vdots & \ddots & \vdots \\
			\sigma_0^2 & \ldots & \sigma_0^2 \\
		\end{pmatrix}
	\end{equation*}
	\textbf{Note:} $\Kmat$ (matrix) $\neq k(\cdot, \cdot)$ (function) --- never write ``$k$ is P.S.D.'' in the exam.
	\medskip

	Prove $\Kmat$ is P.S.D.:
	\begin{enumerate}
		\item \textbf{Prove $\Kmat$ is symmetric}.
		\item \textbf{Prove that $\forall \ \vv \in \mathbb{R}^n, \vv^\top \Kmat \vv \geq 0$.}
	\end{enumerate}
	\end{column}
	\begin{column}{0.24\textwidth}
	\rmcov{1}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{2 (a): How to Prove That A Matrix Is P.S.D.}
\footnotesize
	\begin{enumerate}
		\item<1-> Since $\Kij = \sigma_0^2$ for all $i,j$, we have $\Kmat^\top = \Kmat$, thus $\Kmat$ is symmetric.
		\item<2-> For any $\vv = (v_1, \ldots, v_n)^\top$: show $ \vv^\top \Kmat \vv \geq 0$.
		\begin{enumerate}
			\item<3-> Naive: compute $\Kmat \vv$, then $\vv^\top (\Kmat \vv)$.
			\begin{align*}
				\Kmat \vv &= \sigma_0^2\big(\textstyle\sum_i v_i,\ \sum_i v_i,\ \ldots,\ \sum_i v_i \big)^\top, \\
				\vv^\top \Kmat \vv &= \sigma_0^2 \big[ v_1 (\textstyle\sum_i v_i) + v_2 (\sum_i v_i) + \ldots + v_n (\sum_i v_i) \big] \\
				&= \sigma_0^2 (v_1 + v_2 + \ldots + v_n) (\textstyle\sum_i v_i)
				= \sigma_0^2 \big(\textstyle\sum_i v_i\big)^2 \geq 0.
			\end{align*}
			\item<4-> Faster way: $\Kmat = \sigma_0^2\, \onev \onev^\top$, where $\onev = (1, 1, \ldots, 1)^\top$. So $\vv^\top \Kmat \vv = \sigma_0^2\, \vv^\top \onev \onev^\top \vv = \sigma_0^2 (\onev^\top \vv)^2 \geq 0$.
		\end{enumerate}
	\end{enumerate}
\end{frame}

\begin{frame}[t]{2 (a): Constant Kernel --- Proof via Feature Map}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
	Alternative: write $\kxxp = \sigma_0^2$ as a \textbf{feature-map inner product} $\langle \phi(\xv), \phi(\xvp) \rangle$.

	$\Rightarrow$ construct $\phi(\xv) \in \mathbb{R}^d$ explicitly. Here $\phi(\xv) = \sigma_0$:
	\begin{align*}
		\kxxp & = \sigma_0^2
		= \langle \sigma_0, \sigma_0 \rangle
		= \langle \phi(\xv), \phi(\xvp) \rangle. \hfill\qed
	\end{align*}
	\end{column}
	\begin{column}{0.36\textwidth}
	\centering
	\begin{tikzpicture}[font=\footnotesize, >=Latex,
		blk/.style={draw, rounded corners=3pt, align=center, inner sep=7pt, minimum width=3.1cm, minimum height=1.05cm, fill=green!10, draw=green!45!black}]
	\node[blk] (phi) at (0,0) {feature map\\ $\phi(\xv) = \sigma_0$};
	\node[blk] (k) at (0,-2.1) {$k = \langle\phi(\xv),\phi(\xvp)\rangle$\\ $=\sigma_0^2$};
	\draw[->, thick] (phi) -- (k);
	\end{tikzpicture}
	\\[3pt]
	{\scriptsize An inner product is always a valid kernel.}
	\end{column}
	\end{columns}
\end{frame}

	\begin{align*}
		\kxxp = \underbrace{\sigma_0^2}_{:= k_0(\xv, \xvp)} + \underbrace{\xv^\top \xvp}_{:= k_1(\xv, \xvp)}

		\setlength{\itemsep}{4pt}

\begin{frame}{2 (b): Stationary vs.\ Isotropic --- Overview}
	\small
	\begin{tcolorbox}[colback=blue!4, colframe=blue!55!black, boxsep=2pt, left=5pt, right=5pt, top=3pt, bottom=3pt, arc=2pt]
	\footnotesize
	\textbf{Definitions.} \ $k$ is \textbf{stationary} if $k(\xv, \xv + \dv) = k(\zero, \dv)$ (depends only on the shift $\dv$).\\
	$k$ is \textbf{isotropic} if it is a function of $\|\xv - \xvp\|$ (depends only on the distance).
	\end{tcolorbox}
	\medskip
	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{@{}lccl@{}}
		\toprule
		\textbf{Cov.\ function} & \textbf{Stationary?} & \textbf{Isotropic?} & \textbf{Reason} \\
		\midrule
		Constant $\sigma_0^2$ & \checkmark & \checkmark & const $= \sigma_0^2\|\xv-\xvp\|^0$ \\
		Linear $\sigma_0^2 + \xv^\top\xvp$ & $\times$ & $\times$ & has $\xv^\top\dv$ term \\
		Polynomial $(\sigma_0^2+\xv^\top\xvp)^p$ & $\times$ & $\times$ & same as linear \\
		Squared exp. & \checkmark & \checkmark & function of $\|\xv-\xvp\|$ \\
		Mat\'ern & \checkmark & \checkmark & function of $\|\xv-\xvp\|$ \\
		Exponential & \checkmark & \checkmark & function of $\|\xv-\xvp\|$ \\
		\bottomrule
	\end{tabular}
	\end{center}
\end{frame}

\begin{frame}[t]{2 (b): Constant functions}
	\begin{enumerate}
		\item<1-> $\kxxp = \sigma_0^2$ is stationary since $k(\xv, \xv + \dv) = \sigma_0^2  = k(\zero, \dv)$.
		\item<2-> $\kxxp = \sigma_0^2$ is isotropic since $ \kxxp = \sigma_0^2||\xv - \xvp ||^0$.
	\end{enumerate}
\end{frame}

\begin{frame}[t]{2 (b): $\kxxp = \sigma_0^2 + \xv^\top \xvp$}
	\begin{enumerate}
		\item<1-> $\kxxp = \sigma_0^2 + \xv^\top \xvp$ is NOT stationary, since
		\begin{align*}
			k(\xv, \xv + \dv) &= \sigma_0^2 + \xv^\top \xv + \xv^\top \dv \\
			k(\zero, \dv) &= \sigma_0^2.
		\end{align*}
		\item<2-> It is NOT isotropic, since it cannot be written as a func. of $|| \xv - \xvp||$.
	\end{enumerate}
	\medskip
	\uncover<2->{The polynomial case is analogous --- neither stationary nor isotropic.}
\end{frame}

\begin{frame}[t]{2 (b): $\kxxp = \exp\left( - \frac{||\xv - \xvp||_2^2}{2 \ell^2}\right)$}
	\begin{enumerate}
		\item<1-> $\kxxp = \exp\left( - \frac{||\xv - \xvp||_2^2}{2 \ell^2}\right)$ is stationary, since $k(\xv, \xv + \dv) = \exp \left( - \frac{||\dv||_2^2}{2 \ell^2} \right) = k(\zero, \dv)$.
		\item<2-> It is isotropic, since it is a function of $||\xv - \xvp||$.
	\end{enumerate}
\end{frame}

\begin{frame}[t]{2 (b): Mat\'ern and Exponential Cov. Func.}
	Same argument as the squared-exp kernel --- both are functions of $\|\xv - \xvp\|$, hence \textbf{stationary \& isotropic}:
	\begin{enumerate}
		\item Mat\'ern:
		$$\kxxp = \frac{1}{2^\nu} \left( \frac{\sqrt{2 \nu}}{\ell} || \xv - \xvp|| \right)^\nu K_\nu \left( \frac{\sqrt{2 \nu}}{\ell} || \xv - \xvp|| \right)$$
		\item Exponential:
		$$\kxxp = \exp \left( - \frac{||\xv - \xvp ||}{\ell} \right)$$
	\end{enumerate}
\end{frame}
