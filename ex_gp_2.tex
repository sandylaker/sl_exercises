\begin{frame}{Exercise 1: Gaussian Posterior Process}
Assume your data follows the following law:
\begin{align*}
	\yv = \fv + \epsv, \quad \epsv \sim \N(\zero, \sigma^2 \id),
\end{align*}
with $\fv = \fx \in \R^n$ being a realization of a Gaussian process (GP), for which we a priori assume 
\begin{equation*}
	\fx \sim \gp.
\end{equation*}
$\xv$ is a single (1-D) feature, observed at $n$ points.

(a) Derive / define the prior distribution of $\fv$.
\end{frame}

\begin{frame}{Exercise 1 (b)}
	(b) Derive the posterior distribution of $\fv | \yv$.
\end{frame}

\begin{frame}{1 (b): Think Before Solving}
	\small
	\textbf{Target:}\quad get the posterior $\fv\,|\,\yv$ from likelihood and prior --- without computing any normalizing integral.
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{Both factors are Gaussian.} Bayes gives $p(\fv|\yv)\propto p(\yv|\fv)\,p(\fv)$. A product of two Gaussian densities (in $\fv$) is again Gaussian --- so what kind of function of $\fv$ sits in the exponent?
		\item<2-> \textbf{Complete the square.} The exponent is a quadratic $-\tfrac12\fv^\top A\fv + \fv^\top b$. Which matrix plays the role of $A$, and which vector plays $b$?
		\item<3-> \textbf{Read off the parameters.} Once written as $-\tfrac12(\fv-\mu)^\top A(\fv-\mu)$, the mean and covariance fall out directly. What are they?
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{Gaussian $\times$ Gaussian $\to$ complete the square $\to$ read off $(\fpost,\Kpost)$}}
\end{frame}

\begin{frame}{1 (b): Likelihood and Prior}
\uncover<1->{
	The posterior can be derived from the likelihood and prior using Bayes rule:
	$$\yv = \fv + \epsv, \quad \epsv \sim \N(\zero, \sigma^2 \id)$$
}
\uncover<2->{
The likelihood is:
	\begin{equation*}
		\begin{aligned}
			p(\yv | \fv)
			&\propto \exp \left( -\frac{1}{2} (\yv - \fv)^\top (\sigma^2 \id)^{-1} (\yv - \fv) \right).
		\end{aligned}
	\end{equation*}
}
\uncover<3->{
	The prior is:
	\begin{align*}
		p(\fv) \propto \exp\left( - \frac{1}{2}(\fv - \mv)^\top \Kmat^{-1} (\fv - \mv) \right).
	\end{align*}
}
\end{frame}

\begin{frame}{1 (b): Posterior}
\small
	So the posterior is:
\begin{align*}
	p(\fv | \yv) & \propto p(\yv | \fv) \cdot p(\fv) \\
	\only<2->{
		&\propto \exp\left( -\frac{1}{2} (\yv - \fv)^\top (\sigma^2 \id)^{-1} (\yv - \fv) \right) \cdot \exp\left( - \frac{1}{2}(\fv - \mv)^\top \Kmat^{-1} (\fv - \mv) \right) \\
	}
	\only<3->{
		&\propto \exp \left( -\frac{1}{2} \left(  \fv^\top \ssqiinv\fv - 2 \fv^\top \ssqiinv \yv + \fv^\top \kinv \fv - 2 \fv^\top \kinv \mv \right) \right) \\
		&\propto \exp \left( -\frac{1}{2} \left( \fv^\top (\ssqiinv + \kinv) \fv \  -2\cdot  \fv^\top (\ssqiinv \yv + \kinv \mv) \right) \right) \\
	}
	\only<4->{
		& \propto \exp \bigg( -\frac{1}{2}\fv^\top \underbrace{(\ssqiinv + \kinv)}_{=: \kpinv} \fv \ + \ \fv^\top \underbrace{(\ssqiinv \yv + \kinv \mv)}_{=:\tfv}  \bigg) \\
	}
	\only<5->{
		&\propto \exp \left( - \frac{1}{2} \fv^\top \kpinv \fv + \fv^\top \tfv \right)
	}
\end{align*}
\end{frame}

\begin{frame}{1 (b): Complete the Square}
\small
	\begin{align*}
		p(\fv | \yv) &\propto \exp \left( - \frac{1}{2} \fv^\top \kpinv \fv + \fv^\top \tfv \right)
	\end{align*}
	\begin{tcolorbox}[colback=blue!4, colframe=blue!55!black, boxsep=2pt, left=5pt, right=5pt, top=3pt, bottom=3pt, arc=2pt]
	\footnotesize
	\textbf{Completing the square} (symmetric $\Amat$):
	\begin{itemize}
		\setlength{\itemsep}{2pt}
		\item Scalar: $a x^2 + b x + c = a\big(x + \tfrac{b}{2a}\big)^2 + \big(c - \tfrac{b^2}{4a}\big)$.
		\item Matrix: $\xv^\top \Amat \xv + \xv^\top \bv + c = \big(\xv + \tfrac12 \Amat^{-1}\bv\big)^\top \Amat \big(\xv + \tfrac12 \Amat^{-1}\bv\big) + \big(c - \tfrac14 \bv^\top \Amat^{-1}\bv\big)$.
	\end{itemize}
	\end{tcolorbox}
	\uncover<2->{
		Here $\Amat = -\tfrac{1}{2}\kpinv$, $\bv = \tfv$, $c=0$ (or an arbitrary const). So
		$$\xv + \tfrac{1}{2}\Amat^{-1}\bv = \fv + \tfrac{1}{2}\big(-\tfrac{1}{2}\kpinv\big)^{-1}\tfv = \fv - \Kpost\,\tfv,$$
		and we drop $c - \tfrac14 \bv^\top \Amat^{-1}\bv$ (a constant in $\fv$, absorbed by $\exp$ and $\propto$).
	}
\end{frame}

\begin{frame}{1 (b): Complete the Square}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	With $\Amat = -\tfrac12\kpinv$, the exponent becomes a clean quadratic form in $\fv$:
	\begin{align*}
		p(\fv | \yv) &\propto \exp \bigg( -\tfrac{1}{2}\,(\fv - \Kpost\tfv)^\top \kpinv \,(\fv - \underbrace{\Kpost\tfv}_{:=\,\fpost})\bigg) \\
		&\propto \exp \bigg( -\tfrac{1}{2}\,(\fv - \fpost)^\top \kpinv \,(\fv - \fpost)\bigg).
	\end{align*}
	\uncover<2->{
		This is the kernel of a Gaussian with covariance $\Kpost$ and mean $\fpost$. Hence
		\begin{align*}
			\fv | \yv \sim \N(\fpost, \Kpost). \hfill\qed
		\end{align*}
	}
	\end{column}
	\begin{column}{0.24\textwidth}
	\rmgp{2}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (c)}
	(c) Derive the posterior predictive distribution $\ys | \xs, \xv, \yv$ for a new sample $\xs$ from the \textbf{same} data-generating process.
\end{frame}

\begin{frame}{1 (c): Think Before Solving}
	\small
	\textbf{Target:}\quad the distribution of a \emph{new noisy} target $\ys$ at a test point $\xs$, given the training data.
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{The integral is cumbersome.} Marginalizing $\fv$ out of $p(\ys|\ldots,\fv)\,p(\fv|\yv)$ works but is messy. Is there a shortcut that avoids integrating?
		\item<2-> \textbf{Use the known joint.} Training targets $\yv$ and the new $\ys$ are \emph{jointly} Gaussian. What are the four covariance blocks of $(\yv,\ys)$? Where does the noise $\sigma^2$ enter?
		\item<3-> \textbf{Condition, don't integrate.} The Gaussian conditioning rule turns the joint into $\ys\,|\,\yv$ directly. What are $\muv_{a|b}$ and $\Sigma_{a|b}$ here?
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{known joint Gaussian $(\yv,\ys)$ $\to$ apply conditioning rule $\to$ done (no integral)}}
\end{frame}

\begin{frame}{1 (c): Predictive via Gaussian Conditioning}
	\small
	Na\"ively: marginalize $\fv$ out via $p(\ys|\xs,\xv,\yv)=\int p(\ys|\xs,\xv,\yv,\fv)\,p(\fv|\yv,\xv)\,\mathsf{d}\fv$ --- cumbersome.
	\smallskip

	Instead, use the \textbf{Gaussian conditioning rule}:
	\begin{tcolorbox}[colback=blue!4, colframe=blue!55!black, boxsep=2pt, left=5pt, right=5pt, top=3pt, bottom=3pt, arc=2pt]
	\footnotesize
	If
	$\begin{pmatrix}\av \\ \bv\end{pmatrix}
	\sim \N\!\left(
		\begin{pmatrix}\muv_{a}\\ \muv_{b}\end{pmatrix},
		\begin{pmatrix}\Sigma_{aa} & \Sigma_{ab}\\ \Sigma_{ab}^\top & \Sigma_{bb}\end{pmatrix}
	\right)$,
	then $\av\,|\,\bv \sim \N(\muv_{a|b}, \Sigma_{a|b})$ with
	\begin{align*}
		\muv_{a|b} &= \muv_a + \Sigma_{ab}\,\Sigma_{bb}^{-1}(\bv - \muv_b), &
		\Sigma_{a|b} &= \Sigma_{aa} - \Sigma_{ab}\,\Sigma_{bb}^{-1}\,\Sigma_{ba}.
	\end{align*}
	\end{tcolorbox}
\end{frame}

\begin{frame}[t]{1 (c): Predictive via Gaussian Conditioning}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	\begin{tcolorbox}[colback=blue!10, colframe=blue!10, boxsep=1pt, left=4pt, right=4pt, top=2pt, bottom=2pt, arc=2pt, before skip=2pt, after skip=4pt]
	\footnotesize Here we condition $\ys$ on $\yv$ (not $\fv$), and $\yv=\fv+\epsv$ $\Rightarrow$ add $\sigma^2$ on the diagonal of the cov.\ matrix.
	\end{tcolorbox}

	The joint distribution of $(\yv,\ys)$:
	\begin{align*}
		\begin{pmatrix}\yv \\ \ys\end{pmatrix}
		\sim \N \left(
		\begin{pmatrix}\mv \\ \ms\end{pmatrix},
		\begin{pmatrix}
			\Kmat + \sigma^2 \id & \Kstar \\
			\Kstar^\top  & \Kstarstar + \sigma^2
		\end{pmatrix}
		\right).
	\end{align*}
	\uncover<2->{
		Conditioning ($\av=\ys$, $\bv=\yv$) gives a Gaussian
		$\ys | \xs, \xv, \yv \sim \N(\mu_*, \Sigma_*)$ with (here $\av=\ys$ is \emph{scalar}, so $\Sigma_{a|b}$ is the scalar predictive variance)
		{\footnotesize
		\begin{align*}
			\mu_* &= \ms + \Kstar^\top(\Kmat + \sigma^2 \id)^{-1}(\yv -\mv),\\
			\Sigma_* &= \Kstarstar + \sigma^2 - \Kstar^\top (\Kmat +\sigma^2 \id)^{-1} \Kstar. \hfill\qed
		\end{align*}}%
	}
	\uncover<3->{\scriptsize\textbf{NB:} the variance keeps the $+\sigma^2$ because $\ys$ is the \emph{observable} target. The latent $f_*$ would drop it: $\operatorname{Var}=\Kstarstar - \Kstar^\top(\Kmat+\sigma^2\id)^{-1}\Kstar$.}
	\end{column}
	\begin{column}{0.24\textwidth}
	\rmgp{3}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (d)}
	\small
	(d) Implement the GP with squared exponential (SE) kernel, zero mean function and $\ell = 1$ from scratch for $n=2$ observations $(\yv, \xv)$. Do this as efficiently as possible by explicitly calculating all expensive computations \textbf{by hand}. Do the same for the posterior predictive distribution of $\ys$. Test your implementation using simulated data.

\begin{frame}{1 (d): Think Before Solving}
	\small
	\textbf{Target:}\quad a from-scratch GP for $n=2$ where the "expensive" step is done analytically.
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{9pt}
		\item<1-> \textbf{What is expensive?} The predictive formula needs $(\Kmat+\sigma^2\id)^{-1}$. For $n=2$ this is just a $2\times2$ inverse --- can you write it in closed form?
		\item<2-> \textbf{Few distinct entries.} With the SE kernel and $\ell=1$, $k(x,x)=1$ and off-diagonals share one value $\rho$. How many numbers actually determine $\Kmat$?
		\item<3-> \textbf{Training.} With zero mean and $\ell$ fixed, the only free parameter is $\sigma^2$. How do you pick it from the data (which Gaussian do you maximize)?
	\end{itemize}
	\vspace{2pt}
	\uncover<3->{\centering\recall{$2\times2$ inverse by hand $\to$ plug into part (c) $\to$ tune $\sigma^2$ by marginal likelihood}}
\end{frame}

\begin{frame}{1 (d): Kernel Matrix and Its Inverse by Hand}
	\small
	\begin{columns}[T]
	\begin{column}{0.74\textwidth}
	SE kernel, $\ell=1$: $k(x,x')=\exp\!\big(-\tfrac12(x-x')^2\big)$, so $k(x,x)=1$.
	For $n=2$ set $\rho := \exp\!\big(-\tfrac12(x^{(1)}-x^{(2)})^2\big)\in(0,1]$:
	\begin{align*}
		\Kmat = \begin{pmatrix} 1 & \rho \\ \rho & 1 \end{pmatrix},
		\qquad
		\Kmat^{-1} = \frac{1}{1-\rho^2}\begin{pmatrix} 1 & -\rho \\ -\rho & 1 \end{pmatrix}.
	\end{align*}
	\uncover<2->{
	\textbf{Noisy version} $\Kmaty := \Kmat + \sigma^2\id$ (the matrix the predictive actually inverts):
	{\footnotesize
	\begin{align*}
		\Kmaty = \begin{pmatrix} 1+\sigma^2 & \rho \\ \rho & 1+\sigma^2 \end{pmatrix},
		\quad
		\Kmaty^{-1} = \frac{1}{(1+\sigma^2)^2 - \rho^2}\begin{pmatrix} 1+\sigma^2 & -\rho \\ -\rho & 1+\sigma^2 \end{pmatrix}.
	\end{align*}}%
	}
	\uncover<3->{\scriptsize ($\rho^2 = \exp(-(x^{(1)}-x^{(2)})^2)$.) This closed-form $2\times2$ inverse is the "expensive computation by hand." \hfill\qed}
	\end{column}
	\begin{column}{0.24\textwidth}
	\rmgp{4}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (d): Posterior Predictive at $\xs$ (Zero Mean)}
	\small
	With zero mean ($\mv=\zero$, $\ms=0$) plug $\Kmaty^{-1}$ into the part-(c) formula. For a test point $\xs$,
	$$\Kstar = \big(k(\xs,x^{(1)}),\, k(\xs,x^{(2)})\big)^\top, \qquad \Kstarstar = k(\xs,\xs) = 1.$$
	\begin{align*}
		\text{mean} &= \Kstar^\top \Kmaty^{-1}\, \yv, \\
		\text{var} &= \underbrace{\Kstarstar + \sigma^2}_{=\,1+\sigma^2} - \Kstar^\top \Kmaty^{-1}\, \Kstar.
	\end{align*}
	\uncover<2->{
	\textbf{Training ($\sigma^2$).} Zero mean, $\ell$ fixed $\Rightarrow$ the only free parameter is $\sigma^2$. Maximize the log marginal likelihood of $\yv\sim\N(\zero,\Kmaty)$. Using $\partial\Kmaty/\partial\sigma^2=\id$, set the derivative to zero:
	$$\operatorname{tr}\!\big(\Kmaty^{-1}\yv\yv^\top\Kmaty^{-1} - \Kmaty^{-1}\big) = 0,$$
	and solve for $\sigma^2$ numerically by root-finding. \hfill\qed
	}
\end{frame}

\begin{frame}[fragile]{1 (d): Implementation and Test (Python)}
	\scriptsize
	Pure \texttt{NumPy}/\texttt{SciPy} (no GP library); the prediction uses the part-(c) result.
	\begin{lstlisting}
import numpy as np; from scipy.optimize import brentq
rng = np.random.default_rng(190); n = 2; sigma = 1.0
def kern(x, y=None):                       # SE kernel, l = 1
    d = np.subtract.outer(np.ravel(x), np.ravel(x if y is None else y))
    return np.exp(-0.5 * d**2)             # k(x,y) = exp(-0.5 (x-y)^2)
# simulate noisy targets  y ~ N(0, K + sigma^2 I)
x = rng.standard_normal(n); Ky = kern(x) + sigma**2 * np.eye(n)
y = rng.multivariate_normal(np.zeros(n), Ky).reshape(-1, 1)
# tune sigma^2 by marginal likelihood: solve tr(Kyi yy^T Kyi - Kyi) = 0
def root(s2):
    Kyi = np.linalg.inv(kern(x) + s2 * np.eye(n))
    return 0.5 * np.trace(Kyi @ y @ y.T @ Kyi - Kyi)
s2 = brentq(root, 1e-3, 20)
# posterior predictive at x* = 0  (part (c))
Kyi = np.linalg.inv(kern(x) + s2 * np.eye(n))
Ks = kern(x, [0.0]); Kss = 1.0
mean_pred = Ks.T @ Kyi @ y
var_pred  = Kss + s2 - Ks.T @ Kyi @ Ks     # +s2: observable y*
	\end{lstlisting}
\end{frame}
