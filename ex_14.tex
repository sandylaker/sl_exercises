\begin{frame}{Exercise 1: Entropy}
	A fair die is rolled at the same time as a fair coin is tossed. Let $A$ be the number on the upper surface of the dice and let $B$ describe the outcome of the coin toss, where 
	\begin{align*}
		B = \begin{cases}
			1, \quad & \text{head,} \\
			0 \quad & \text{tail.}
		\end{cases}
	\end{align*}
	Two random variables $X$ and $Y$ are given by $X = A + B$ and $Y = A - B$, respectively.
	\vspace{10pt}
	
	(a) Calculate the entropies $H(X)$ and $H(Y)$, the conditional entropies $H(Y|X)$ and $H(X|Y)$, the joint entropy $H(X, Y)$ and the mutual information $I(X;Y)$.
\end{frame}

\begin{frame}{Exercise 1 (a): Setup}
\small
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}
\begin{columns}[T,totalwidth=\textwidth]
	\begin{column}{0.48\textwidth}
		\textbf{Definitions}
		\[
			X = A+B, \qquad Y = A-B
		\]

		\textbf{Elementary probabilities}
		\[
		\begin{aligned}
			P(A=a) &= \frac{1}{6},\quad a=1,\ldots,6, \\
			P(B=b) &= \frac{1}{2},\quad b=0,1, \\
			P(A=a,B=b) &= \frac{1}{12}.
		\end{aligned}
		\]

		\textbf{Before $H(X)$ and $H(Y)$}
		
		Compute the joint distribution of $X$ and $Y$ by using the relation between
		$(x,y)$ and $(a,b)$.
	\end{column}

	\begin{column}{0.48\textwidth}
		\uncover<2->{
		\textbf{One-to-one joint mapping}
		\[
		(a,b) \mapsto (x,y) = (a+b, a-b)
		\]
		\[
		(x,y) \mapsto (a,b) =
		\left(\frac{x+y}{2}, \frac{x-y}{2}\right)
		\]
		}

		\uncover<3->{
		\textbf{Consequence}
		\[
			P(X=x,Y=y) = P(A=a,B=b) = \frac{1}{12}
		\]
		for every feasible $(x,y)$.
		}
	\end{column}
\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (b)}
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\textbf{Question.} For independent discrete random variables $X$ and $Y$, show
\[
	I(X; X + Y) - I(Y; X + Y) = H(X) - H(Y).
\]

\only<2>{
\textbf{Solution.}
\[
\begin{aligned}
	I(X; X+Y) - I(Y; X+Y)
	&= H(X+Y) - H(X+Y|X) - H(X+Y) + H(X+Y|Y).
\end{aligned}
\]
}

\only<3->{
\textbf{Solution.}
\[
\begin{aligned}
	I(X; X+Y) - I(Y; X+Y)
	&= H(X+Y) - H(X+Y|X) - H(X+Y) + H(X+Y|Y) \\
	&= H(X+Y|Y) - H(X+Y|X).
\end{aligned}
\]
}

\begin{columns}[T,totalwidth=\textwidth]
		\begin{column}{0.42\textwidth}
			\only<4->{
			\textbf{Conditional entropy step}
			\[
				p(x+y|x) = p(y|x)
			\]
			Given $x$, knowing $x+y$ is the same as knowing $y$:
			\[
				H(X+Y|X) = H(Y|X).
			\]
			Similarly,
			\[
				H(X+Y|Y) = H(X|Y).
			\]
			}
		\end{column}
	\begin{column}{0.58\textwidth}
		\only<5->{
			\textbf{Remaining calculation}
		}
		\only<5>{
			\[
			\begin{aligned}
				& H(X+Y|Y) - H(X+Y|X)
			\end{aligned}
			\]
		}
		\only<6>{
			\[
			\begin{aligned}
				& H(X+Y|Y) - H(X+Y|X) \\
				&= H(X|Y) - H(Y|X)
			\end{aligned}
			\]
		}
		\only<7->{
			\[
			\begin{aligned}
				& H(X+Y|Y) - H(X+Y|X) \\
				&= H(X|Y) - H(Y|X) \\
				&= H(X) - H(Y),
			\end{aligned}
			\]
		because $X$ and $Y$ are independent.
		}
	\end{column}
\end{columns}
\end{frame}

\begin{frame}{Exercise 3: Smoothed Cross-Entropy Loss}
	\textit{Over-confidence}: model is more confident in its prediction than the data warrants. \textbf{Label smoothing} (Szegedy et al., 2016) alleviates this and improves robustness.

	Conventional cross-entropy minimizes the KL-divergence between the ground-truth distribution $d$ (a delta: $d_k = 1$ for the true class) and the predicted distribution $\pi(\mathbf{x} | \bm{\theta})$.

	Label smoothing smooths $d$: given $\beta$ (e.g.\ $\beta = 0.1$), distribute probability mass $\beta$ uniformly across the $g$ classes and reduce the ground-truth mass:
	\begin{align*}
		\tilde{d}_k = \begin{cases}
			\beta / g & \text{for } d_k = 0, \\
			1 - \beta + \beta / g & \text{for } d_k = 1.
		\end{cases}
	\end{align*}
	The smoothed cross-entropy is then $D_{KL}(\tilde{d} \, \| \, \pi(\mathbf{x}|\bm{\theta}))$.

	\vspace{4pt}
	\textbf{(a)} Derive the empirical risk when using the smoothed cross-entropy as loss function. \textit{Hint: some terms can be merged into a constant and ignored during implementation.}
\end{frame}

\begin{frame}{Exercise 3 (a): $D_{KL}(\tilde{d} \, \| \, \pi(\mathbf{x}|\bm{\theta}))$}
\begin{columns}[T,totalwidth=\textwidth]
	\begin{column}{0.68\textwidth}
		\small
		Start from the KL definition and expand:
		\begin{align*}
			R_{\text{emp}}(\bm{\theta})
			&= \frac{1}{n} \sum_{i=1}^n \sum_{k=1}^g \tilde{d}_k^{(i)} \log \frac{\tilde{d}_k^{(i)}}{\pi_k(\mathbf{x}^{(i)} | \bm{\theta})} \\
			&= \frac{1}{n} \sum_{i=1}^n \sum_{k=1}^g \Big( \underbrace{\tilde{d}_k^{(i)} \log \tilde{d}_k^{(i)}}_{\text{indep.\ of } \bm{\theta}} - \tilde{d}_k^{(i)} \log \pi_k(\mathbf{x}^{(i)} | \bm{\theta}) \Big).
		\end{align*}
	\end{column}
	\begin{column}{0.30\textwidth}
		\uncover<1->{
		\begin{tcolorbox}[
			title=Recall: $\tilde{d}_k$,
			colback=white,
			colframe=violet,
			fonttitle=\bfseries,
			boxrule=0.8pt,
			arc=2pt,
			left=2pt, right=2pt, top=2pt, bottom=2pt
		]
		\scriptsize
		\[
			\tilde{d}_k = \begin{cases}
				\beta / g, & d_k = 0 \\
				1 \!-\! \beta \!+\! \beta / g, & d_k = 1
			\end{cases}
		\]
		\end{tcolorbox}
		}
	\end{column}
\end{columns}

\vspace{28pt}

\uncover<2->{
\small
Drop the first term (constant w.r.t.\ $\bm{\theta}$) and plug in $\tilde{d}_k$ (with $y^{(i)}$ the true class) to get the practical form:
\begin{align*}
	R_{\text{emp}}(\bm{\theta}) = - \frac{1}{n} \sum_{i=1}^n \left[ (1 - \beta) \log \pi_{y^{(i)}}(\mathbf{x}^{(i)} | \bm{\theta}) + \frac{\beta}{g} \sum_{k=1}^g \log \pi_k(\mathbf{x}^{(i)} | \bm{\theta}) \right] + \text{const}.
\end{align*}
}
\end{frame}

\begin{frame}[fragile]{Exercise 3 (b): Implementation (Python)}
	\small
	\textbf{(b)} Implement the smoothed cross-entropy. Vectorised \texttt{numpy} version:

	\vspace{4pt}
	\begin{lstlisting}[style=Pystyle]
import numpy as np
from numpy.typing import NDArray


def smoothed_ce_loss(label: NDArray,    # shape (n,), values in {1, ..., g}
                     pred: NDArray,     # shape (n, g), predicted probs
                     smoothing: float) -> float:
    assert 0.0 <= smoothing <= 1.0
    n, g = pred.shape
    assert label.shape == (n,)

    # Build smoothed label matrix: start uniform, set true class to 1 - b + b/g
    base = smoothing / g
    smoothed = np.full((n, g), base)
    smoothed[np.arange(n), label - 1] = 1.0 - smoothing + base

    # Per-sample cross-entropy, then mean over the batch
    return float(-np.sum(smoothed * np.log(pred), axis=1).mean())
	\end{lstlisting}
\end{frame}
