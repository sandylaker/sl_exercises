\begin{frame}{Exercise 1: Risk Minimizers for 0-1-Loss}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
	\textbf{Setting.} Classification with $\Yspace = \{1, \ldots, g\}$ and $\Hspace = \{h: \Xspace \rightarrow \Yspace \}$. The loss of interest is the 0-1-loss:
	$$L(y, h(\xv)) = \id_{y \neq h(\xv)} = \begin{cases}
		1, & \text{if } y \neq h(\xv), \\
		0, & \text{if } y = h(\xv).
	\end{cases}
	$$

	\vspace{2pt}
	\textbf{Question (a).} Take the hypothesis space of \emph{constant models}
	$$\Hspace = \{h \mid h(\xv) = \thetav \in \Yspace \ \forall \xv \in \Xspace \}.$$
	Show that
	$$\hat{h}(\xv) = \mathrm{mode} \left\{\yi \right\}$$
	is the empirical risk minimizer for the 0-1-loss.
\end{column}
\begin{column}{0.46\textwidth}
	\uncover<2->{
		\textbf{Roadmap} {\footnotesize (general recipe for ERM w/ const.\ models)}\\[2pt]
		{\color{gray}\hrule height 0.4pt}\vspace{4pt}
		\begin{enumerate}
			\item Write down the empirical risk $\riske(h)$.
			\item Substitute $h(\xv) = \thetav$ (hypothesis is constant).
			\item \textcolor{gray}{Check strict convexity in $\thetav$.} \\
				  \textcolor{gray}{\footnotesize (not needed here)}
			\item \textcolor{gray}{Set the gradient to $0$ and solve for $\thetav^\ast$.} \\
				  \textcolor{gray}{\footnotesize (not needed here)}
		\end{enumerate}
	}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Solution (a) --- Steps 1 \& 2: write $\riske(h)$, then substitute $h(\xv) = \thetav$}
\small
Recall the 0-1-loss:
$$L(y, h(\xv)) = \id_{y \neq h(\xv)} = \begin{cases}
	1, & \text{if } y \neq h(\xv), \\
	0, & \text{if } y = h(\xv).
\end{cases}$$

Step 1 --- Write down the empirical risk.
\vspace{-6pt}
\begin{align*}
	\riske(h) &= \sumin \id_{\yi \neq h(\xi)}.
\end{align*}

\uncover<2->{
\vspace{-8pt}
Step 2 --- Substitute $h(\xv) = \thetav$ (constant model):
\vspace{-6pt}
\begin{align*}
	\riske(\thetav) &= \sumin \id_{\yi \neq \thetav}.
\end{align*}
}

\uncover<3->{
\vspace{-4pt}
\emph{Intuition:}
\begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}
	\item $\riske(\thetav)$ = \# training points whose label $\neq \thetav$.
	\item To minimize errors, pick $\thetav$ = most frequent $\yi$, i.e.\ the \textbf{mode}.
\end{itemize}
}
\end{frame}

\begin{frame}{Solution (a) --- Instantiating the intuition}
\small
\begin{columns}[T]
\begin{column}{0.55\textwidth}
Formalize ``pick the most frequent $\yi$'' as an argmin chain:
\begin{align*}
	\argmin_{\thetav \in \Yspace} \riske(\thetav)
	&= \argmin_{\thetav \in \Yspace} \sumin \id_{\yi \neq \thetav} \\
	\only<2->{&= \argmin_{\thetav \in \Yspace} \sumin \left(1 - \id_{\yi = \thetav}\right) \\}
	\only<3->{&= \argmax_{\thetav \in \Yspace} \sumin \id_{\yi = \thetav} \\}
	\only<4->{&= \mathrm{mode}\left\{ \yi \right\}.}
\end{align*}
\uncover<4->{
Therefore $\hat{h}(\xv) = \mathrm{mode}\left\{ \yi \right\}$ is the empirical risk minimizer.
}
\end{column}
\begin{column}{0.43\textwidth}
\emph{Toy example:} $n = 5$ points, labels $\{1,1,1,0,0\}$.
\vspace{2pt}

\begin{frame}{Question (b)}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
	\textbf{Question (b).} What is the optimal constant model in terms of the \emph{theoretical} risk for the 0-1-loss, and what is its risk?
	\vspace{6pt}

	\textbf{Recall.} Constant model:
	$$\Hspace = \{h \mid h(\xv) = \thetav \in \Yspace \ \forall \xv \in \Xspace \}.$$

	0-1-loss:
	$L(y, h(\xv)) = \id_{y \neq h(\xv)}$.
\end{column}
\begin{column}{0.46\textwidth}
	\uncover<2->{
		\textbf{Roadmap} {\footnotesize (same recipe, now for \emph{theoretical} risk)}\\[2pt]
		{\color{gray}\hrule height 0.4pt}\vspace{4pt}
		\begin{enumerate}
			\item Write down the theoretical risk $\risk_L(h)$.
			\item Substitute $h(\xv) = \thetav$ (hypothesis is constant).
			\item Marginalize out $\xv$ (integrand no longer depends on $\xv$).
			\item Minimize in $\thetav$ $\Rightarrow$ optimal $\thetav^\ast$; then plug back to get $\risk_L(\bar h)$.
		\end{enumerate}
	}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Solution (b) --- Steps 1--3: write $\risk_L$, substitute, marginalize out $\xv$}
	\small
	\setlength{\abovedisplayskip}{0pt}
	\setlength{\belowdisplayskip}{0pt}
	\setlength{\abovedisplayshortskip}{0pt}
	\setlength{\belowdisplayshortskip}{0pt}
	\vspace*{-1.2em}
	\begin{align*}
		\risk_L(h)
		&= \int_y \int_\xv \id_{y \neq h(\xv)}\, p(\xv, y)\, \mathrm{d}\xv\, \mathrm{d}y
		&& \text{\scriptsize (Step 1: write the theoretical risk)} \\
		\only<2->{&= \int_y \int_\xv \id_{y \neq \thetav}\, p(\xv, y)\, \mathrm{d}\xv\, \mathrm{d}y
		&& \text{\scriptsize (Step 2: substitute $h(\xv) = \thetav$)} \\}
		\only<3->{&= \int_y \id_{y \neq \thetav}\, p(y)\, \mathrm{d}y
		&& \text{\scriptsize (Step 3: integrand is $\xv$-free, marginalize)}}
	\end{align*}

\end{frame}

\begin{frame}{Solution (b) --- Step 4: minimize in $\thetav$}
	\small
	\setlength{\abovedisplayskip}{0pt}
	\setlength{\belowdisplayskip}{0pt}
	\setlength{\abovedisplayshortskip}{0pt}
	\setlength{\belowdisplayshortskip}{0pt}
	\vspace*{-1.2em}

	From Step 3:
	\[
		\risk_L(\thetav)
		= \int_y \id_{y \neq \thetav}\,p(y)\,\mathrm{d}y
		= \mathbb{E}[\id_{y \neq \thetav}]
		= \Prob(y \neq \thetav),
		\qquad \mathbb{E}[\id_A] = \Prob(A).
	\]

	\uncover<2->{
	Use the complement:
	\[
		\risk_L(\thetav)
		= \Prob(y \neq \thetav)
		= 1 - \Prob(y = \thetav).
	\]
	}

	\uncover<3->{
	Minimizing risk therefore means choosing the most likely label:
	\begin{align*}
		\argmin_{\thetav \in \Yspace} \risk_L(\thetav)
		&= \argmin_{\thetav \in \Yspace} \left(1 - \Prob(y = \thetav)\right) \\
		&= \argmax_{l \in \Yspace} \Prob(y = l).
	\end{align*}
	}
\end{frame}

\begin{frame}{Solution (b) --- Step 4: plug back to get $\risk_L(\bar h)$}
	\small

	From Step~4, the optimal constant model is
	\[
		\bar h(\xv) \;=\; \argmax_{l \in \Yspace} \Prob(y = l),
	\]
	i.e.\ the label that attains the maximum of $\Prob(y = l)$, so
	\[
		\Prob\!\left(y = \bar{h}(\xv)\right) \;=\; \max_{l \in \Yspace} \Prob(y = l).
	\]

	\uncover<2->{
	Plug into $\risk_L(\thetav) = 1 - \Prob(y = \thetav)$ (from Step~4):
	\[
		\risk_L(\bar h)
		\;=\; 1 - \Prob\!\left(y = \bar{h}(\xv)\right)
		\;=\; 1 - \max_{l \in \Yspace} \Prob(y = l).
	\]
	}
\end{frame}

\begin{frame}{Summary of (b) --- General takeaways}
	\small

	\begin{itemize}
		\setlength{\itemsep}{8pt}

		\item \textbf{ERM and theoretical risk minimization share one recipe} (for constant models) --- write, substitute, simplify, minimize. The only difference is how you average:
		\[
			\underbrace{\tfrac{1}{n}\sumin L(\yi, h(\xi))}_{\riske(h)\ \text{(sample)}}
			\;\;\longleftrightarrow\;\;
			\underbrace{\E_{\xv, y}\!\left[L(y, h(\xv))\right]}_{\risk_L(h)\ \text{(population)}}.
		\]
		Here that yields $\;\text{mode}\{\yi\}\;\longleftrightarrow\;$ mode of $p(y)$.

		\item \textbf{The loss selects the statistic} you end up estimating. \\
		$0\text{-}1$ loss $\Rightarrow$ \emph{mode}; \quad L2 loss $\Rightarrow$ \emph{mean}; \quad L1 loss $\Rightarrow$ \emph{median}.

		\item \textbf{Indicator trick:} $\E[\id_A] = \Prob(A)$. Turns $0\text{-}1$-loss integrals into probabilities --- keeps the algebra clean and makes the answer interpretable.

		\item \textbf{Hypothesis class $\Hspace$ controls what you can exploit.} \\
		Constant models drop $\xv$ $\Rightarrow$ only the \emph{marginal} $p(y)$ matters. A richer $\Hspace$ that depends on $\xv$ would let you target $p(y \mid \xv)$ pointwise (Bayes classifier).
	\end{itemize}
\end{frame}

\begin{frame}{Question (c)}
	\small

	\textbf{Question (c).} Derive the approximation error if the hypothesis space $\Hspace$ consists of the \emph{constant models}.
	\vspace{8pt}

	\begin{enumerate}
		\setlength{\itemsep}{6pt}
		\item How is the approximation error defined?
		\uncover<2->{\\[2pt]
			\textcolor{gray}{$\displaystyle\inf_{h \in \Hspace} \risk_L(h) \;-\; \riskbayes_L$}
		}

		\item What do the two terms mean?
		\uncover<3->{\\[2pt]
			\textcolor{gray}{$\inf_{h \in \Hspace} \risk_L(h)$: best risk \emph{inside} $\Hspace$. \quad $\riskbayes_L$: best risk over \emph{all} $h$ (Bayes-optimal).}
		}

		\item Which $\Hspace$ are we using here?
		\uncover<4->{\\[2pt]
			\textcolor{gray}{Constant models --- same setup as (a) and (b).}
		}

	\end{enumerate}
\end{frame}

\begin{frame}{Question (c) --- Using (a) and (b) for both terms}
	\small

	\emph{First term (from (b)):}
	\[
		\inf_{h \in \Hspace} \risk_L(h) \;=\; 1 - \max_{l \in \Yspace} \Prob(y = l).
	\]

	\uncover<2->{
	\emph{Second term --- tower property:}
	\[
		\riskbayes_L
		\;=\; \E_{\xv, y}\!\left[\id_{y \neq h^\ast(\xv)}\right]
		\;=\; \E_\xv\!\left[\,\E_{y \mid \xv}\!\left[\id_{y \neq h^\ast(\xv)} \,\big|\, \xv\right]\right].
	\]
	}

	\uncover<3->{
	Minimize the inner expectation pointwise --- this is (b) with $p(y \mid \xv)$ in place of $p(y)$:
	\[
		\E_{y \mid \xv}\!\left[\id_{y \neq h^\ast(\xv)} \,\big|\, \xv\right] \;=\; 1 - \max_{l \in \Yspace} \Prob(y = l \mid \xv).
	\]
	}

	\uncover<4->{
	\emph{Take expectation over $\xv$:}
	\[
		\riskbayes_L \;=\; 1 - \E_\xv\!\left[\max_{l \in \Yspace} \Prob(y = l \mid \xv)\right].
	\]
	}
\end{frame}

\begin{frame}{Solution to (c)}
	\begin{align*}
		\inf_{h \in \Hspace} \risk_L(h) - \riskbayes_L 
		&= \risk_L(\bar{h}) - \riskbayes_L \\
		&= (1 - \max_{l \in \Yspace} \Prob(y=l)) - (1 - \Ex[\max_{l \in \Yspace} \Prob(y = l | \xv)]) \\
		&= \Ex[\max_{l \in \Yspace} \Prob(y = l | \xv)] - \max_{l \in \Yspace} \Prob(y=l).
	\end{align*}
\end{frame}

\begin{frame}{Question (d)}

Furthermore, consider the probabilistic 0-1-loss

\begin{frame}{Solution to Question (d) --- Setup}
	\small
	\begin{columns}[T]
	\begin{column}{0.55\textwidth}

		\[
			\begin{array}{c|cc}
				 & y = 0 & y = 1 \\ \hline

		\]

		\[
			L(y, \pix) = {\color{red!70!black}\id_{\pix \geq 1/2}\,\id_{y=0}} + {\color{blue!70!black}\id_{\pix < 1/2}\,\id_{y=1}}.
		\]
	\end{column}
	\begin{column}{0.43\textwidth}
		\uncover<2->{

		}
		\vspace{10pt}

		\uncover<3->{
			\textbf{Step 3: Tower property.}

		}

\begin{frame}{Solution to Question (d) --- Compute $\Eyx[L \mid \xv]$}
	\setlength{\abovedisplayskip}{4pt}
	\setlength{\belowdisplayskip}{4pt}
	\begin{columns}[T]
	\begin{column}{0.66\textwidth}
		\begin{flalign*}
			& \Eyx[L(y, \pix) \ | \ \xv] & \\
			&= \Eyx[\id_{\pix \geq 1/2} \id_{y=0} + \id_{\pix < 1/2} \id_{y=1} \ | \ \xv] & \\
			\only<2->{&= \Eyx[\id_{\pix \geq 1/2} \id_{y=0} \ | \ \xv] + \Eyx[\id_{\pix < 1/2} \id_{y=1} \ | \ \xv] & \\}
			\only<3->{&= \id_{\pix \geq 1/2} \cdot \Eyx[\id_{y=0} \ | \ \xv] + \id_{\pix < 1/2} \cdot \Eyx[\id_{y=1} \ | \ \xv] & \\}
			\only<4->{&= \id_{\pix \geq 1/2}\, \Prob(y = 0 \ | \ \xv) + \id_{\pix < 1/2}\, \Prob(y = 1 \ | \ \xv). &}
		\end{flalign*}
	\end{column}
	\begin{column}{0.32\textwidth}
		\uncover<3->{
			\footnotesize
			\textbf{Why} $\Eyx[\id_{y=l} \mid \xv] = \Prob(y = l \mid \xv)$?\\[4pt]
			An indicator only takes values $\{0,1\}$:
			\[
				\id_{y=l} = \begin{cases} 1, & y = l, \\ 0, & y \neq l. \end{cases}
			\]
			So its conditional expectation is
			\begin{align*}
				\Eyx[\id_{y=l} \mid \xv]
				&= 1 \cdot \Prob(y=l \mid \xv) \\
				&\phantom{={}} + 0 \cdot \Prob(y \neq l \mid \xv) \\
				&= \Prob(y = l \mid \xv).
			\end{align*}
		}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Solution to Question (d) --- Minimizer \& (non-)uniqueness}
	Combining both cases, any minimizer satisfies
	$$
		\pix = \begin{cases}
			< 1/2, \quad &\text{if $\Prob(y = 0 \ | \ \xv) \geq 1 / 2$,} \\
			\geq 1/2, \quad &\text{if $\Prob(y=0 \ | \ \xv) < 1/2$.}
		\end{cases}
	$$
	\uncover<2->{
		\vspace{4pt}
		The posterior $p_{y|\xv}(1 \mid \xv)$ is naturally of this form --- but it is \emph{not} the only $\pi$ of this kind: any choice on the correct side of $1/2$ works.
	}
	\uncover<3->{
		\vspace{4pt}
		\textbf{Conclusion (i):} the minimizer is \textbf{not unique}.
	}
\end{frame}

\begin{frame}{Solution to Question (d) --- Pros \& cons}
