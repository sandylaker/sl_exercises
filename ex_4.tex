\begin{frame}{Exercise 4: Multiclass Hinge Loss}
	\small
	Consider the multiclass classification scenario consisting of a feature space $\Xspace$ and a label space $\Yspace = \{1, \ldots, g\}$ with $g \geq 2 $ classes. Moreover, we consider the hypothesis space of models based on $g$ discriminant/scoring functions:
	$$\Hspace = \{ f = (f_1, \ldots, f_g)^T: \Xspace \to \mathbb{R}^g | f_k: \Xspace \to \mathbb{R},  \ \forall k \in \Yspace \}.$$
	A model $f$ in $\Hspace$ is used to make a prediction by means of transforming the scores into classes by choosing the class with the maximum score:
	\begin{align}
		h(\xv) = \argmax_{k \in \{1, \ldots, g \}} f_k(\xv).
		\label{eq:1}
	\end{align} 
	The multiclass hinge loss is defined by $$L(y, \fx) = \max_k (\fkx - f_y(\xv) + \id_{y \neq k}).$$
	(a) Show that 0-1-loss for a predictor $h$ as in (\ref{eq:1}) based on a model $f \in \Hspace$ is at most the multiclass hinge loss for $f$ i.e., $$L_{0-1}(y, \hx) = \id_{y\neq \hx} \leq L(y, \fx).$$
\end{frame}

\begin{frame}{Digest the loss \& roadmap for (a)}
	\small
	\textbf{Read the hinge loss term by term:}
	$$L(y, \fx) = \max_k\big(\fkx - \fyx + \id_{y \neq k}\big).$$
	\begin{itemize}
		\item<1-> $\fkx - \fyx$: score gap of $k$ over true class $y$.
		\item<2-> $\id_{y \neq k}$: $+1$ if $k \neq y$, $0$ if $k = y$.
		\item<3-> $\max_k$: take the worst offender.
		\item<4-> $k = y$ gives term $= 0$, so $L \geq 0$ \emph{always}.
	\end{itemize}

	\vspace{6pt}
	\uncover<5->{\textbf{What (a) asks.}\ 0-1 loss is $0$ (correct) or $1$ (wrong). Want $L(y, \fx) \geq L_{0\text{-}1}(y, \hx)$ in both regimes.}

	\vspace{6pt}
	\uncover<6->{\textbf{Roadmap: split by whether $\argmax$ matches $y$.}}
	\begin{itemize}
		\item<6-> Case 1, $y = \argmax_k \fkx$: pick the trivial term $k = y$ inside the max; compare against $L_{0\text{-}1}$.
		\item<7-> Case 2, $y \neq \argmax_k \fkx$: pick the offender $k^\star = \argmax_k \fkx$ inside the max; compare against $L_{0\text{-}1}$.
	\end{itemize}
\end{frame}

\begin{frame}{Solution to (a) --- Case 1: $y = \argmax_k \fkx$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		Prediction correct, so
		\begin{align*}
			L_{0\text{-}1}(y, \hx) = 0.
		\end{align*}

		\vspace{4pt}
		\uncover<2->{
			Plug $k = y$ into the max:
			$$\fkx - \fyx + \id_{y \neq k} = 0 - 0 + 0 = 0.$$
			Hence
			\begin{align*}
				L(y, \fx) = \max_k \big(\fkx - \fyx + \id_{y \neq k}\big) \geq 0.
			\end{align*}
		}

		\vspace{4pt}
		\uncover<3->{
			$\Rightarrow L(y, \fx) \geq 0 = L_{0\text{-}1}(y, \hx).$ \quad $\checkmark$
		}
	\end{column}
	\begin{column}{0.36\textwidth}
		\footnotesize
		\begin{tcolorbox}[
			colback=cyan!8, colframe=cyan!55!blue,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=0-1 loss, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L_{0\text{-}1}(y, \hx) = \id_{y \neq \hx}$
		\end{tcolorbox}

		\vspace{4pt}
		\begin{tcolorbox}[
			colback=orange!10, colframe=orange!70!red,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Multiclass hinge, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L(y, \fx) = \max_k\!\big(\fkx - \fyx + \id_{y \neq k}\big)$
		\end{tcolorbox}

		\vspace{4pt}
		Predictor:\ $\hx = \argmax_k \fkx$.
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Solution to (a) --- Case 2: $y \neq \argmax_k \fkx$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		Prediction wrong, so
		\begin{align*}
			L_{0\text{-}1}(y, \hx) = 1.
		\end{align*}

		\vspace{4pt}
		\uncover<2->{
			Pick $k^\star = \argmax_k \fkx$. Then $f_{k^\star}(\xv) > \fyx$ and $\id_{y \neq k^\star} = 1$:
			$$\underbrace{f_{k^\star}(\xv) - \fyx}_{> 0} + \underbrace{\id_{y \neq k^\star}}_{= 1} > 1.$$
			Hence
			\begin{align*}
				L(y, \fx) = \max_k \big(\fkx - \fyx + \id_{y \neq k}\big) > 1.
			\end{align*}
		}

		\vspace{4pt}
		\uncover<3->{
			$\Rightarrow L(y, \fx) > 1 = L_{0\text{-}1}(y, \hx).$ \quad $\checkmark$
		}

		\vspace{4pt}
		\uncover<4->{
			\textbf{Both cases combined:}\ $L_{0\text{-}1}(y, \hx) = \id_{y \neq \hx} \leq L(y, \fx).$ \quad $\blacksquare$
		}
	\end{column}
	\begin{column}{0.36\textwidth}
		\footnotesize
		\begin{tcolorbox}[
			colback=cyan!8, colframe=cyan!55!blue,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=0-1 loss, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L_{0\text{-}1}(y, \hx) = \id_{y \neq \hx}$
		\end{tcolorbox}

		\vspace{4pt}
		\begin{tcolorbox}[
			colback=orange!10, colframe=orange!70!red,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Multiclass hinge, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L(y, \fx) = \max_k\!\big(\fkx - \fyx + \id_{y \neq k}\big)$
		\end{tcolorbox}

		\vspace{4pt}
		Predictor:\ $\hx = \argmax_k \fkx$.
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Question (b)}
	(b) Verify that the multiclass hinge loss of $f \in \Hspace$ on a data point $(\xv, y) \in \Xspace \times \Yspace$ is bounded from above by $\sum_{k\neq y} \max \{0, 1 + \fkx - f_y(\xv) \}$.
	\vspace{10pt}
	\emph{Hint}: Note that this upper bound is sometimes referred to as the multiclass hinge loss.
\end{frame}

\begin{frame}{Digest (b) \& how to reuse (a)}
	\small
	\begin{columns}[T]
	\begin{column}{0.58\textwidth}
		\textbf{Difference from (a).}
		\begin{itemize}
			\item<1-> (a): hinge upper-bounds 0-1.
			\item<2-> (b): \emph{sum surrogate} upper-bounds hinge.
			\item<3-> Same direction; swap $\max_k$ for $\sum_{k\neq y}\max\{0,\cdot\}$.
		\end{itemize}

		\vspace{8pt}
		\uncover<4->{\textbf{Reuse (a)'s case split.}}
		\begin{itemize}
			\item<4-> Same case split as (a), via $k^\star = \argmax_k \fkx$.
			\item<5-> From (a): in case 2, the max in $L$ is attained at $k^\star$.
			\item<6-> That $k^\star$-summand already sits inside RHS; the remaining summands are $\geq 0$, so RHS $\geq L$.
		\end{itemize}
	\end{column}
	\begin{column}{0.4\textwidth}
		\footnotesize
		\begin{tcolorbox}[
			colback=orange!10, colframe=orange!70!red,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Multiclass hinge, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L(y, \fx) = \max_k\!\big(\fkx - \fyx + \id_{y \neq k}\big)$
		\end{tcolorbox}

		\vspace{4pt}
		\begin{tcolorbox}[
			colback=green!10, colframe=green!45!black,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Sum surrogate, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$\displaystyle\sum_{k\neq y}\max\{0, 1 + \fkx - \fyx\}$
		\end{tcolorbox}

		\vspace{4pt}
		\uncover<7->{%
		\begin{tcolorbox}[
			colback=cyan!8, colframe=cyan!55!blue,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Sandwich (a)+(b), fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			\centering
			$L_{0\text{-}1} \,\overset{\text{(a)}}{\leq}\, L \,\overset{\text{(b)}}{\leq}\, \displaystyle\sum_{k\neq y}\max\{0,\cdot\}$\\[2pt]
			{\scriptsize \emph{Smooth surrogate for 0-1.}}
		\end{tcolorbox}}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Solution to (b) --- Case 1: $y = k^\star$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		If $k^\star = y$, plug $k = k^\star = y$ into the max:
		$$L(y, \fx) = f_{k^\star}(\xv) - \fyx + 0 = 0.$$

		\vspace{6pt}
		\uncover<2->{
			RHS is a sum of $\max\{0,\cdot\} \geq 0$, hence non-negative:
			$$L(y, \fx) = 0 \,\leq\, \sum_{k \neq y}\max\{0, 1 + \fkx - \fyx\}. \quad\checkmark$$
		}
	\end{column}
	\begin{column}{0.36\textwidth}

			boxsep=2pt
		]
			$L(y, \fx) = \max_k\!\big(\fkx - \fyx + \id_{y \neq k}\big)$
		\end{tcolorbox}

		\vspace{4pt}

\end{frame}

\begin{frame}{Solution to (b) --- Case 2: $y \neq k^\star$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		From (a): $L(y, \fx) = f_{k^\star}(\xv) - \fyx + 1 > 1$.

		\vspace{4pt}
		\uncover<2->{
		\begin{align*}
			L(y, \fx)
			&= \underbrace{f_{k^\star}(\xv) - \fyx + 1}_{\geq 0} \\
			&= \max\{0,\, f_{k^\star}(\xv) - \fyx + 1\} \\
			&\leq \max\{0, f_{k^\star}(\xv) - \fyx + 1\} \\
			&\qquad + \underbrace{\textstyle\sum_{k \neq y,\, k^\star}\max\{0, \fkx - \fyx + 1\}}_{\geq 0} \\
			&= \sum_{k \neq y}\max\{0,\, 1 + \fkx - \fyx\}. \quad\checkmark
		\end{align*}
		}

		\vspace{4pt}
		\uncover<3->{
			\textbf{Both cases combined:}\ $L(y, \fx) \leq \sum_{k \neq y}\max\{0,\, 1 + \fkx - \fyx\}$. \quad$\blacksquare$
		}
	\end{column}
	\begin{column}{0.36\textwidth}
		\footnotesize
		\begin{tcolorbox}[
			colback=orange!10, colframe=orange!70!red,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Multiclass hinge, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$L(y, \fx) = \max_k\!\big(\fkx - \fyx + \id_{y \neq k}\big)$
		\end{tcolorbox}

		\vspace{4pt}
		\begin{tcolorbox}[
			colback=green!10, colframe=green!45!black,
			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Sum surrogate, fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$\displaystyle\sum_{k\neq y}\max\{0, 1 + \fkx - \fyx\}$
		\end{tcolorbox}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Question (c)}
	In the case of binary classification, i.e., $g=2$ and $\Yspace = \{-1, +1\}$, we use a single discriminant model $\fx = f_1(\xv) - f_{-1}(\xv)$ based on two scoring functions: $f_1, f_{-1}: \Xspace \to \mathbb{R}$ for the prediction by means of $\hx = \mathrm{sgn}(\fx)$. Here, $f_1$ is the score for the positive class and $f_{-1}$ is the score for the negative class. Show that the upper bound in (b) coincide with the binary hinge loss $L(y, \fx) = \max \{0, 1 - y \fx \}$.
\end{frame}

\begin{frame}{Digest (c) \& how to reuse (b)}
	\small
	\begin{columns}[T]
	\begin{column}{0.58\textwidth}

			\item<1-> Specialize: $g = 2$, $\Yspace = \{-1, +1\}$.
			\item<2-> Single discriminant: $\fx = f_1(\xv) - f_{-1}(\xv)$.

\begin{frame}{Solution to (c) --- Case 1: $y = +1$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		\begin{align*}
			& \sum_{k \neq y}\max\{0, 1 + \fkx - \fyx\} \\
			&= \sum_{k \neq +1}\max\{0,\, \fkx - f_1(\xv) + 1\} \\
			&= \max\{0,\, \underbrace{f_{-1}(\xv) - f_1(\xv)}_{=\, -\fx} + 1\} \\
			&= \max\{0,\, 1 - \fx\} \\
			&= \max\{0,\, 1 - y\fx\}. \quad\checkmark
		\end{align*}
	\end{column}
	\begin{column}{0.36\textwidth}

			title=Starting point (from (b)), fonttitle=\footnotesize\bfseries,
			boxsep=2pt
		]
			$\displaystyle\sum_{k\neq y}\max\{0, 1 + \fkx - \fyx\}$
		\end{tcolorbox}

		\vspace{4pt}

			boxrule=0.5pt, arc=2pt,
			left=4pt, right=4pt, top=2pt, bottom=2pt,
			title=Target: binary hinge, fonttitle=\footnotesize\bfseries,

			$\max\{0,\, 1 - y\fx\}$

		\vspace{4pt}

\begin{frame}{Solution to (c) --- Case 2: $y = -1$}
	\small
	\begin{columns}[T]
	\begin{column}{0.62\textwidth}
		\begin{align*}
			& \sum_{k \neq y}\max\{0, 1 + \fkx - \fyx\} \\
			&= \sum_{k \neq -1}\max\{0,\, \fkx - f_{-1}(\xv) + 1\} \\

			&= \max\{0,\, 1 + \fx\} \\

\begin{frame}{Question (d)}
	\small
	Recall the statement of the lecture regarding the binary hinge loss:

	\vspace{4pt}
	\emph{``... the hinge loss only equals zero for a margin $\geq 1$ encouraging confident (correct) predictions.''}

	\vspace{6pt}
	Can we say something similar for the alternative multiclass hinge loss in (b)?

	\vspace{4pt}
	\emph{Hint:} multiclass hinge loss: $\sum_{k\neq y} \max\{0, 1 + \fkx - f_y(\xv)\}$, and all $\max\{0, \cdot\}$ terms are non-negative.

	\vspace{8pt}
	\textbf{Think about:}
	\begin{itemize}
		\item ``Similar'' = an iff for when the loss vanishes. Hypothesize:\quad multiclass hinge $= 0 \;\iff\;$ \emph{???}
		\item Binary: 1 margin, condition ``$\geq 1$''. Multiclass: how many margins per data point, and what replaces ``$\geq 1$''?
	\end{itemize}
\end{frame}

\begin{frame}{Solution to Question (d)}
	Yes, we can say that \emph{it is only zero if all the $g-1$ margins are greater than 1.}
	\begin{itemize}
		\item<2-> Margins: $m_{y, k}(\xv) = \fyx - \fkx$, where $k \in \Yspace \setminus \{y\}$.
		\item<3-> Mathematically: $ m_{y, k}(\xv) \geq 1 \ \forall k \neq y  \Leftrightarrow \sum_{k \neq y} \max \{0, 1 + \fkx - \fyx \} = 0$.
	\end{itemize}
	
	\vspace{10pt}
	\uncover<4->{
		Proof: 
		\begin{align*}
			m_{y, k}(\xv) \geq 1 \ \forall k \neq y  
			&\Rightarrow \fyx - \fkx  \geq 1 \ \forall k \neq y \\
			&\Rightarrow \fkx - \fyx \leq -1 \ \forall k \neq y \\
			&\Rightarrow \max \{0, 1 + \fkx - \fyx \} = 0 \ \forall k \neq y \\
			&\Rightarrow \sum_{k \neq y} \max \{0, 1 + \fkx - \fyx \} = 0
		\end{align*}
	}
\end{frame}

\begin{frame}[fragile]{Code (Python): one-vs-rest}
	\scriptsize
	\begin{columns}[T]
	\begin{column}{0.5\textwidth}
		\begin{lstlisting}[style=Pystyle, basicstyle=\ttfamily\fontsize{6pt}{7pt}\selectfont]
import numpy as np
from numpy.typing import NDArray
from scipy.optimize import minimize


# Vectorised empirical risk for binary hinge.
# Z = [X | y] of shape (n, p + 1); margins is (n,).
# All rows are handled in one matmul + elementwise op.
def emp_risk_bin_hinge(theta: NDArray,
                       Z: NDArray) -> float:
    X, y = Z[:, :p], Z[:, p]
    margins = y * (X @ theta)
    return float(np.maximum(0.0, 1.0 - margins).sum())
		\end{lstlisting}
	\end{column}
	\begin{column}{0.5\textwidth}
		\begin{lstlisting}[style=Pystyle, basicstyle=\ttfamily\fontsize{6pt}{7pt}\selectfont]
# One-vs-rest fitting: g INDEPENDENT optims.
# Output: theta_mat of shape (p, g);
# column i = parameters of the i-th binary classifier.
def one_vs_all_bin_theta(Z: NDArray) -> NDArray:
    theta_mat = np.zeros((p, g))
    for i in range(1, g + 1):
        # codebook in one vectorised step
        y_rec = np.where(Z[:, p] == i, 1.0, -1.0)
        Z_coded = np.column_stack([Z[:, :p], y_rec])
        theta_mat[:, i - 1] = minimize(
            emp_risk_bin_hinge,
            np.zeros(p),
            args=(Z_coded,)).x
    return theta_mat


# Prediction: pick class with max score.
def one_vs_all_bin_predict(theta_mat: NDArray,
                           x: NDArray) -> int:
    return int(np.argmax(x @ theta_mat)) + 1
		\end{lstlisting}
	\end{column}
	\end{columns}

\begin{frame}[fragile]{Code (Python): multiclass (joint)}
	\scriptsize
	\begin{columns}[T]
	\begin{column}{0.5\textwidth}
		\begin{lstlisting}[style=Pystyle, basicstyle=\ttfamily\fontsize{6pt}{7pt}\selectfont]
import numpy as np
from numpy.typing import NDArray
from scipy.optimize import minimize


# Vectorised empirical risk for the multiclass sum surrogate.
# Shapes: Theta is (p, g); scores (n, g); f_y (n,); margins (n, g).
# The k=y entry of `margins` is always 1, so its max(0, .) = 1
# per row; subtract n_loc to recover the sum over k != y only.
def emp_risk_multiclass(Theta_flat: NDArray,
                        Z: NDArray) -> float:
    Theta   = Theta_flat.reshape(g, p).T
    X, y    = Z[:, :p], Z[:, p].astype(int)
    scores  = X @ Theta
    n_loc   = X.shape[0]
    f_y     = scores[np.arange(n_loc), y - 1]
    margins = 1.0 + scores - f_y[:, None]
    return float(np.maximum(0.0, margins).sum() - n_loc)
		\end{lstlisting}
	\end{column}
	\begin{column}{0.5\textwidth}
		\begin{lstlisting}[style=Pystyle, basicstyle=\ttfamily\fontsize{6pt}{7pt}\selectfont]
# Multiclass fitting: ONE joint optim over all p*g params.
# Output: Theta of shape (p, g). All columns are coupled
# by the joint optimisation.
def multiclass_theta(Z: NDArray) -> NDArray:
    Theta_flat = minimize(emp_risk_multiclass,
                          np.zeros(p * g),
                          args=(Z,)).x
    return Theta_flat.reshape(g, p).T


# Prediction: same argmax rule as one-vs-rest.
# Difference is only in how Theta is fit.
def multiclass_predict(Theta: NDArray,
                       x: NDArray) -> int:
    return int(np.argmax(x @ Theta)) + 1
		\end{lstlisting}
	\end{column}
	\end{columns}
\end{frame}

		\\\hline
		\textbf{(c)} For $g = 2$, $\fx = f_1 - f_{-1}$: the sum surrogate reduces to the binary hinge $\max\{0, 1 - y\fx\}$.
		&
