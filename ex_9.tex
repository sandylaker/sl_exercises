\begin{frame}{Exercise 1: Kernelized Multiclass SVM}
	\small
	For a data set $\D = \{(\xv^{(1)}, y^{(1)}), \ldots, (\xv^{(n)}, y^{(n)}) \}$ with $\yi \in \Yspace = \{+1, -1 \}$, assume that we are provided with a suitable feature map $\phi: \Xspace \to \Phi$, where $\Phi \subset \mathbb{R}^d$. In the featureized SVM learning problem we are facing the following optimization problem:
	\begin{align*}
		\min_{\thetab, \theta_0, \sli} &\frac{1}{2} \thetab^\top \thetab + C \sumin \sli \\
		\text{s.t. } & \yi \left(\left\langle \thetab, \phixi \right\rangle  + \theta_0\right) \geq 1 - \sli \qquad \forall i \in \{1, \ldots, n \}, \\
		\text{and } & \sli \geq 0 \qquad i \in \{1, \ldots, n \},
	\end{align*}
	where $C \geq 0$ is some constant.
	
	(a) Argue that this is equivalent to the following ERM problem:
		\begin{align*}
			\riske(\thetab) = \frac{1}{2} ||\thetab||^2 + C\sumin \max(1 - \yi(\thetab^\top \phixi + \theta_0), 0).
		\end{align*}
		i.e., the regularized ERM problem for the hinge loss for the hypothesis space 
		\begin{align*}
			\Hspace = \{f: \Phi \to \mathbb{R} \ | \ f(\zv) = \thetab^\top \zv + \theta_0, \thetab \in \mathbb{R}^d, \theta_0 \in \mathbb{R} \}
		\end{align*}
\end{frame}

\begin{frame}{1(a): Rewriting the Optimization Target}
	\textbf{Optimization target:}
	\begin{align*}
	    \min_{\thetab, \theta_0, \zeta^{(i)}} &\frac{1}{2} \thetab^\top \thetab + C \sum_{i=1}^n \zeta^{(i)} \\
	    \text{s.t. } & \sli \geq 1 - y^{(i)} \left(\langle \thetab, \phi(\mathbf{x}^{(i)}) \rangle + \theta_0 \right), \quad \forall i, \\
	    \text{and } & \zeta^{(i)} \geq 0, \quad \forall i.
	\end{align*}
	
\end{frame}


\begin{frame}{1(a): Comparison between Optimization Target and $\mathcal{R}_{\text{emp}}$}
\small
	\textbf{Optimization target:}
	\begin{align*}
	    \min_{\thetab, \theta_0, \zeta^{(i)}} &\frac{1}{2} \thetab^\top \thetab + C \sum_{i=1}^n \zeta^{(i)} \\
	    \text{s.t. } & \sli \geq 1 - y^{(i)} \left(\langle \thetab, \phi(\mathbf{x}^{(i)}) \rangle + \theta_0 \right), \quad \forall i, \\
	    \text{and } & \zeta^{(i)} \geq 0, \quad \forall i.
	\end{align*}
	
	\uncover<2->{
    \textbf{Empirical risk:}
    \begin{align*}
        \mathcal{R}_{\text{emp}}(\thetab) = \frac{1}{2} \|\thetab\|^2 + C \sum_{i=1}^n \max(1 - y^{(i)}(\thetab^\top \phi(\mathbf{x}^{(i)}) + \theta_0), 0). \quad
    \end{align*}
    }
    
    \uncover<3->{
    \textbf{Observation:} Both contain $\frac{1}{2} \thetab^\top \thetab$. Both contain $C \sumin \ldots$ penalty terms.
    
    \textbf{Next:} Prove that $\sli$ equals $\max(\ldots)$ term.
    }
\end{frame}

\begin{frame}{1 (a): Prove $\sli$ Equals $\max(\ldots)$ Term}
	\small
	\uncover<1->{
	For each $i$, The constraints in the optimization problem:
	\begin{align*}
		\sli &\geq \underbrace{1 - y^{(i)} \left(\langle \thetab, \phi(\mathbf{x}^{(i)}) \rangle + \theta_0 \right)}_{\text{(i)}} \\
		\sli &\geq \underbrace{0}_{\text{(ii)}}
	\end{align*}
	$\sli \geq$ (i) and (ii) $\Rightarrow$ $\sli \geq $ the larger term in (i) and (ii) $\Rightarrow$ $\sli \geq \max(\text{(i)}, \text{(ii)})$.
	}
	
	\vspace{10pt}
	
	\uncover<2->{
	Therefore, the constraints translate to $\sli \geq \max\left(1 - y^{(i)} \left(\thetab^\top \phi(\mathbf{x}^{(i)}) + \theta_0 \right), 0 \right)$
	}
	\vspace{10pt}
	
	\uncover<3->{
	\textbf{Note:} Our target is to $\min_{\thetab, \theta_0, \zeta^{(i)}} \frac{1}{2} \thetab^\top \thetab + C \sum_{i=1}^n \zeta^{(i)}$, so smaller $\sli$ are preferred. $\leadsto$ Choose $ \sli = \max(\ldots, 0) $
	}

	\uncover<3->{
	\begin{flushright}
	\setlength{\fboxsep}{6pt}%
	\colorbox{blue!10}{\footnotesize recall SVM Part 1, Ex.~1(b)}
	\end{flushright}
	}
	
\end{frame}

\begin{frame}{1 (a): Choose $\sli = \max(\ldots, 0) $ }
The optimization target 

$$
\frac{1}{2} \thetab^\top \thetab + C \sumin \sli
$$
becomes 

$$
\frac{1}{2} \thetab^\top \thetab + C \sumin \max\left(1 - y^{(i)} \left(\thetab^\top \phi(\mathbf{x}^{(i)}) + \theta_0 \right), 0 \right)
$$

which is exactly $\riske$. \qed
\end{frame}


\begin{frame}{Exercise 1 (b)}
	(b) Now assume we deal with a multiclass classification problem with a data set $\D = \{(\xv^{(1)}, y^{(1)}), \ldots, (\xv^{(n)}, y^{(n)})\}$ such that $y^{(i)} \in \Yspace = \{1, \ldots, g \}$ for each $i \in \{1, \ldots, n \}$. In this case, we can derive a similar regularized ERM problem by using the multiclass hinge loss (see Exercise Sheet 4(b)):
	$$\risket = \frac{1}{2} || \thetab||^2 + C \sumin \sum_{y\neq \yi} \max (1 + \thetabt^\top \psi (\xi, y) - \thetabt^\top \psi(\xi, \yi), 0), $$
	where $\thetabt := (\theta_0, \thetab^\top)^\top \in \mathbb{R}^{d + 1}$, and $\psi: \Xspace \times \Yspace \to \mathbb{R}^{d+1}$ is a suitable (multiclass) feature map. Specify a $\psi$ such that this regularized multiclass ERM problem coincides with the regularized binary ERM problem in (a).
\end{frame}

\begin{frame}{1 (b): Think Before Solving}
	\small
	\[ \text{(a) binary:}\quad \riske(\thetab) = \tfrac{1}{2}\|\thetab\|^2 + C\sumin \max(1 - \yi(\thetab^\top \phixi + \theta_0), 0) \]
	\[ \text{(b) multiclass:}\quad \risket = \tfrac{1}{2}\|\thetab\|^2 + C \sumin \sum_{y\neq \yi} \max(1 + \thetabt^\top \psi(\xi, y) - \thetabt^\top \psi(\xi, \yi), 0) \]
	\rule{\linewidth}{0.4pt}
	\vspace{4pt}
	\begin{itemize}
		\setlength{\itemsep}{8pt}
		\item<1-> \textbf{Multiclass.} SVM is binary by design --- one-vs-one, one-vs-rest, or one joint loss?
		\item<2-> \textbf{Intercept.} Margin needs $+\,\theta_0$, but $\psi$ gives only an inner product $\langle \thetabt, \psi(\xi, y) \rangle$. How to absorb $\theta_0$?
		\item<3-> \textbf{Equivalence.} Why do (a) and (b) coincide in the binary case ($g = 2$)?
	\end{itemize}
\end{frame}

\begin{frame}{1(b): A Closer Look at the Multiclass Hinge Loss}
	\small
	\begin{columns}[T]
	\begin{column}{0.7\textwidth}
	\textbf{Goal:} multiclass hinge $\to$ binary hinge when $g = 2$.
	\medskip
	\begin{itemize}
		\setlength{\itemsep}{10pt}
		\item<1-> Binary encoding: $\yi \in \{1,2\} \;\mapsto\; \{-1,+1\}$.
		\item<2-> Then $\sum_{y\neq\yi}$ collapses to \textbf{one term} ($y = -\yi$).
		\item<3-> $\psi(\xi,\yi)$ uses \emph{both} $\xi$ and $\yi$ --- unlike $\phi(\xi)$.
		\item<4-> No $\theta_0$ in the multiclass loss $\Rightarrow$ absorb it how? \textit{(next)}
	\end{itemize}
	\end{column}
	\begin{column}{0.26\textwidth}
	\qroadmap{1}
	\end{column}
	\end{columns}
	
\end{frame}

\begin{frame}{1(b): Define $\psi: \Xspace \times \Yspace \rightarrow \mathbb{R}^{d+1}$}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	\textbf{Goal:} write the margin $y(\thetab^\top \phi(\xv) + \theta_0)$ as one inner product.
	\medskip
	\begin{enumerate}
		\setlength{\itemsep}{10pt}
		\item<2-> Absorb $\theta_0$ via a dummy feature $1$:
		\[ \thetab^\top \phi(\xv) + \theta_0 = \underbrace{\begin{pmatrix} \theta_0 \\ \thetab \end{pmatrix}^{\!\top}}_{\thetabt^\top} \underbrace{\begin{pmatrix} 1 \\ \phi(\xv) \end{pmatrix}}_{\tilde\phi(\xv)} \]
		\item<3-> Pull in $y$:\quad $y\,\thetabt^\top \tilde\phi(\xv) = \thetabt^\top \big(y\,\tilde\phi(\xv)\big)$
		\item<4-> Scale by $\tfrac12$:\quad $\boxed{\;\psi(\xv, y) = \tfrac{1}{2}\, y\, \tilde\phi(\xv)\;}$
	\end{enumerate}
	\end{column}
	\begin{column}{0.25\textwidth}
	\qroadmap{2}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (b): What We Prove Next}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	With $\psi(\xv, y) = \tfrac12\, y\, \tilde\phi(\xv)$, it remains to show --- in the binary case ($g = 2$) --- that the multiclass penalty reduces to the binary hinge of (a):
	\medskip
	\[ \underbrace{\sum_{y\neq \yi} \max(1 + \thetabt^\top \psixiy - \thetabt^\top \psixiyi, 0)}_{\text{(b) multiclass}} \]
	\[ =\ \underbrace{\max(1 - \yi (\thetab^\top \phixi + \theta_0), 0)}_{\text{(a) binary}}. \]
	\end{column}
	\begin{column}{0.25\textwidth}
	\qroadmap{3}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (b): Reduce to the Binary Hinge}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	Binary case $\Rightarrow$ the sum has \textbf{one term} ($y = -\yi$). Then:
	\uncover<2->{
	\begin{align*}
		& 1 + \thetabt^\top \psixiy - \thetabt^\top \psixiyi \\
		&= 1 + \tfrac{1}{2} y\, \thetabt^\top \phitxi - \tfrac{1}{2} \yi\, \thetabt^\top \phitxi \\
		&= 1 + \tfrac{1}{2}(y - \yi)\, \thetabt^\top \phitxi \\
		&= \begin{cases} 1 + \thetabt^\top \phitxi, & \yi = -1 \\ 1 - \thetabt^\top \phitxi, & \yi = +1 \end{cases} \\
		&= 1 - \yi\, \thetabt^\top \phitxi.
	\end{align*}
	}
	\end{column}
	\begin{column}{0.25\textwidth}
	\qroadmap{3}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Solution to 1 (b): Continued}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	From the per-term reduction, the sum collapses to a single term:
	\begin{align*}
		\risket &= \tfrac{1}{2}\|\thetab\|^2 + C \sumin \max(1 - \yi\, \thetabt^\top\phitxi, 0) \\
		&= \tfrac{1}{2}\|\thetab\|^2 + C \sumin \max(1 - \yi (\thetab^\top \phixi + \theta_0), 0).
	\end{align*}
	\uncover<2->{\colorbox{blue!10}{multiclass ERM $=$ binary ERM (a)}\;\qed}
	\end{column}
	\begin{column}{0.25\textwidth}
	\qroadmap{3}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 1 (c)}
	\footnotesize
	Show that the regularized multiclass ERM problem in (b) can be written in the kernelized form:
	$$\frac{1}{2} \betav^\top \Kmat \betav + C\sumin \sum_{y \neq \yi} \max (1 + (\Kmat \betav)_{(i-1)g + y} - (\Kmat \betav)_{(i-1)g + \yi}, 0),$$
	where $\betav \in \mathbb{R}^{ng}$ and $\Kmat = \Xmat \Xmat^\top$ for $\Xmat \in \mathbb{R}^{ng \times (d+1)}$ with row entries $\psi(\xi, y)^\top $ for $i = 1, \ldots, n$, $y = 1, \ldots, g$, i.e.,

	\medskip
	\begin{columns}[T]
	\begin{column}{0.44\textwidth}
	$$\Xmat = \begin{pmatrix}
		\psi(\xv^{(1)}, 1)^\top \\
		\psi(\xv^{(1)}, 2)^\top \\
		\vdots \\
		\psi(\xv^{(1)}, g)^\top \\
		\psi(\xv^{(2)}, 1)^\top \\
		\vdots \\
		\psi(\xv^{(n)}, g)^\top \\
	\end{pmatrix}.$$

	Here, $(\Kmat \betav)_{(i-1)g + y}$ is the $((i-1)g + y)$-th entry of $\Kmat \betav$.
	\end{column}
	\begin{column}{0.56\textwidth}
	\textbf{How to get there?}
	\begin{enumerate}
		\setlength{\itemsep}{7pt}
		\item \emph{Representer theorem:} $\thetab^* \in \mathrm{span}\{\psi(\xi, y)\}$. \\ Rewrite $\thetab$ with $\Xmat$ and $\betav$?
		\item Then express $\tfrac12\|\thetab\|^2$ via $\Kmat = \Xmat\Xmat^\top$ and $\betav$?
		\item And the scores $\thetab^\top \psi(\xi, y)$ via $\Kmat\betav$?
	\end{enumerate}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{1 (c): Express $\|\thetab\|^2$ with $\Kmat$ and $\betav$}
	\small
	\begin{columns}[T]
	\begin{column}{0.72\textwidth}
	Representer theorem: $\thetab^*$ is a linear combination of the $\psi(\xi, y)$, i.e.
	\[ \thetab = \Xmat^\top \betav, \qquad \betav \in \mathbb{R}^{ng}, \]
	with the $\psi(\xi, y)^\top$ as the rows of $\Xmat$ (as on the previous slide).
	\uncover<2->{
	\medskip
	With $\Kmat = \Xmat \Xmat^\top$:
	\[ \|\thetab\|^2 = \thetab^\top \thetab = (\Xmat^\top \betav)^\top \Xmat^\top \betav = \boxed{\,\betav^\top \Kmat \betav\,}. \]
	}
	\end{column}
	\begin{column}{0.25\textwidth}
	\qroadmapc{2}
	\end{column}
	\end{columns}
\end{frame}

\begin{frame}{Exercise 2: Kernel Trick}
	The polynomial kernel is defined as 
	$$k(x, \xt) = (x^\top \xt + b)^d.$$
	Furthermore, assume that $x \in \mathbb{R}^2$ and $d = 2$.
	(a) Derive the explicit feature map $\phi$ taking into account that the following equation holds:
	$$k(x, \xt) = \langle \phi(x), \phi(\xt) \rangle$$
	
\end{frame}

\begin{frame}{Solution to 2 (a)}
	\begin{align*}
		k(x, \xt) &= (x^\top \xt + b)^2 = 
		\left( 
			\begin{pmatrix}
				x_1 \\
				x_2
			\end{pmatrix}^\top 
			\begin{pmatrix}
				\xt_1 \\
				\xt_2 \\
			\end{pmatrix}
			+ b 
		\right)^2 \\
		\only<2->{
			&= (x_1 \xt_1 + x_2 \xt_2 + b)^2 \\}
		\only<3->{
			&= x_1^2 \xt_1^2 + 2 x_1 \xt_1 x_2 \xt_2 + x_2^2 \xt_2^2 + 2b x_1 \xt_1 + 2b x_2 \xt_2 + b^2 \\}
		\only<4->{
			&= \left\langle 
			\begin{pmatrix}
				x_1^2 \\
				\sqrt{2} x_1 x_2 \\
				x_2^2 \\
				\sqrt{2b} x_1 \\
				\sqrt{2b} x_2 \\
				b \\
			\end{pmatrix}, 
			\begin{pmatrix}
				\xt_1^2 \\
				\sqrt{2} \xt_1 \xt_2 \\
				\xt_2^2 \\
				\sqrt{2b} \xt_1 \\
				\sqrt{2b} \xt_2 \\
				b \\
			\end{pmatrix} 
			\right\rangle \\
			&= \langle \phi(x), \phi(\xt) \rangle
		}
	\end{align*}
\end{frame}
